  2++ 02B4              ; PROJECT 124 — 8086 physical memory backend for NeoGS GS105a
  3++ 02B4              ;
  4++ 02B4              ; x86 physical $00000-$FFFFF maps to NeoGS RAM pages 4..67
  5++ 02B4              ; (NeoGS physical base $010000, page size $4000 / 16 KiB).
  6++ 02B4              ; Window C000-FFFF is switched through MPAGEX. This keeps firmware in
  7++ 02B4              ; window 0 / page 0 and the current DMA frame in window 1 / page 3.
  8++ 02B4              ;
  9++ 02B4              ; x86_mem_addr is a 20-bit little-endian address:
 10++ 02B4              ;   +0 = bits 7:0, +1 = bits 15:8, +2 = bits 19:16 (valid 0..0F).
 11++ 02B4              ;
 12++ 02B4              ; API:
 13++ 02B4              ;   x86_mem_init: enable NOROM + EXPAG, preserving other GSCFG0 bits.
 14++ 02B4              ;   x86_mem_read8:  IN x86_mem_addr; OUT A=value, carry=invalid address.
