  2++ 02A6              ; PROJECT 124 — 8086 physical memory backend for NeoGS GS105a
  3++ 02A6              ;
  4++ 02A6              ; x86 physical $00000-$FFFFF maps to NeoGS RAM pages 4..67
  5++ 02A6              ; (NeoGS physical base $010000, page size $4000 / 16 KiB).
  6++ 02A6              ; Window C000-FFFF is switched through MPAGEX. This keeps firmware in
  7++ 02A6              ; window 0 / page 0 and the current DMA frame in window 1 / page 3.
  8++ 02A6              ;
  9++ 02A6              ; x86_mem_addr is a 20-bit little-endian address:
 10++ 02A6              ;   +0 = bits 7:0, +1 = bits 15:8, +2 = bits 19:16 (valid 0..0F).
 11++ 02A6              ;
 12++ 02A6              ; API:
 13++ 02A6              ;   x86_mem_init: enable NOROM + EXPAG, preserving other GSCFG0 bits.
 14++ 02A6              ;   x86_mem_read8:  IN x86_mem_addr; OUT A=value, carry=invalid address.
 15++ 02A6              ;   x86_mem_write8: IN x86_mem_addr and x86_mem_value; carry=invalid address.
 16++ 02A6              ;   x86_mem_test: destructive test of the reserved 1 MiB x86 RAM region.
 17++ 02A6              ;                 status=0, AX=0001 on pass; status=5, AX=phase:page on fail.
 18++ 02A6              ;
 19++ 02A6              ; The test leaves NOROM+EXPAG enabled. Only window 3 is switched by MPAGEX.
 20++ 02A6              ; =========================================================================
 21++ 02A6
 22++ 02A6              X86_MEM_NGSPAGE_BASE EQU 4
 23++ 02A6              X86_MEM_PAGE_COUNT   EQU 64
