; -------------------------------------------------------------------------
; upload_firmware
; IN : HL = custom firmware binary, DE = byte length.
; OUT: Carry clear only after custom firmware sends NGS_FIRMWARE_READY_ACK.
;
; Cold/warm reset path:
;   #F4 cold restart
;   #20 get total RAM, read three bytes to prove stock ROM is ready
;   #14 receive executable bootstrap directly into RAM at $4000
;   #13 jump to $4000
;   #A1 bootstrap ready, send length, receive #A3 length accepted
;   send custom firmware bytes to RAM $0000, receive #A2 firmware ready
; -------------------------------------------------------------------------
upload_firmware::
        ld      (neo_fw_source), hl
        ld      (neo_fw_length), de

        ; Firmware image must fit in low RAM $0000-$2FFF, below SP=$3FFF.
        ld      a, d
        or      e
        jp      z, .upload_failed
        ld      a, d
        cp      #30
        jr      c, .firmware_length_valid
        jp      nz, .upload_failed
        ld      a, e
        or      a
        jp      nz, .upload_failed
.firmware_length_valid:
        xor     a
        ld      (neo_fw_bytes_sent), a
        ld      (neo_fw_bytes_sent+1), a
        ld      (neo_fw_timeout_phase), a
        in      a, (GSSTAT)
        ld      (neo_fw_initial_status), a
        ld      (neo_fw_last_status), a

        ; Phase 1: documented GS105a cold restart. Also returns a running
        ; Project 124 firmware to the stock ROM before the new bootstrap.
        ld      a, 1
        ld      (neo_fw_timeout_phase), a
        ld      a, GS_CMD_COLD_RESTART
        call    gs_boot_send_command_long
        jp      c, .upload_failed

        ; Phase 2: #20 returns three RAM-size bytes only after stock ROM is up.
        ; Retry #F4 once: if the previous run was interrupted during DMA, the
        ; first #F4 may have been consumed as the DMA end marker; the second
        ; reaches the command dispatcher after it returns to its main loop.
        ld      a, 2
        ld      (neo_fw_timeout_phase), a
        call    gs_boot_probe_rom
        jr      nc, .stock_rom_ready
        ld      a, 1
        ld      (neo_fw_timeout_phase), a
        ld      a, GS_CMD_COLD_RESTART
        call    gs_boot_send_command_long
        jp      c, .upload_failed
        ld      a, 2
        ld      (neo_fw_timeout_phase), a
        call    gs_boot_probe_rom
        jp      c, .upload_failed
.stock_rom_ready:

        ; Phase 3: use the native GS105a memory loader, not the MOD player.
        ; #14 RECEIVE BLOCK expects the first (negative length low) byte
        ; before the command, then negative length high, address low/high,
        ; and exactly N payload bytes. The payload is written directly to RAM.
        ld      a, 3
        ld      (neo_fw_timeout_phase), a
        ld      hl, 0
        ld      de, neogs_bootstrap_end - neogs_bootstrap_start
        or      a
        sbc     hl, de                  ; HL = -bootstrap length
        ld      (neo_fw_block_length), hl
        ld      a, l
        call    gs_boot_write_command_argument
        jp      c, .upload_failed
        ld      a, GS_CMD_RECEIVE_BLOCK
        call    gs_boot_send_command
        jp      c, .upload_failed

        ; Remaining #14 header: -length high, destination $4000.
        ld      a, (neo_fw_block_length+1)
        call    gs_boot_send_data_byte
        jp      c, .upload_failed
        xor     a
        call    gs_boot_send_data_byte  ; destination low = $00
        jp      c, .upload_failed
        ld      a, #40
        call    gs_boot_send_data_byte  ; destination high = $40
        jp      c, .upload_failed

        ; Phase 4: transfer the executable bootstrap into RAM $4000.
        ld      a, 4
        ld      (neo_fw_timeout_phase), a
        ld      hl, neogs_bootstrap_start
        ld      de, neogs_bootstrap_end - neogs_bootstrap_start
.bootstrap_block_loop:
        ld      a, d
        or      e
        jr      z, .bootstrap_block_done
        ld      a, (hl)
        call    gs_boot_send_data_byte
        jp      c, .upload_failed
        inc     hl
        dec     de
        jr      .bootstrap_block_loop
.bootstrap_block_done:

        ; Phase 5: #13 JUMP TO ADDRESS. The low byte is the command's
        ; first argument and must be written before issuing the command.
        ld      a, 5
        ld      (neo_fw_timeout_phase), a
        xor     a                       ; entry address low = $00
        call    gs_boot_write_command_argument
        jp      c, .upload_failed
        ld      a, GS_CMD_JUMP_ADDRESS
        call    gs_boot_send_command
        jp      c, .upload_failed
        ld      a, #40                  ; entry address high = $40
        call    gs_boot_send_data_byte
        jp      c, .upload_failed

        ; Phase 6: bootstrap has enabled RAM-over-ROM and is waiting at $0000.
        ld      a, 6
        ld      (neo_fw_timeout_phase), a
        call    gs_boot_receive_data_byte
        jp      c, .upload_failed
        cp      NGS_BOOTSTRAP_READY_ACK
        jp      nz, .upload_failed

        ; Phase 7: send firmware length low/high and require ACK #A3.
        ld      a, 7
        ld      (neo_fw_timeout_phase), a
        ld      hl, (neo_fw_length)
        ld      a, l
        call    gs_boot_send_data_byte
        jp      c, .upload_failed
        ld      a, h
        call    gs_boot_send_data_byte
        jp      c, .upload_failed
        call    gs_boot_receive_data_byte
        jp      c, .upload_failed
        cp      NGS_BOOTSTRAP_LENGTH_ACK
        jp      nz, .upload_failed

        ; Phase 8: stream the custom firmware to RAM address $0000.
        ld      a, 8
        ld      (neo_fw_timeout_phase), a
        ld      hl, (neo_fw_source)
        ld      de, (neo_fw_length)
.firmware_stream_loop:
        ld      a, d
        or      e
        jr      z, .firmware_stream_done
        ld      a, (hl)
        call    gs_boot_send_data_byte
        jp      c, .upload_failed
        inc     hl
        dec     de
        jr      .firmware_stream_loop

.firmware_stream_done:
        ; Phase 9: new firmware has jumped to $0000 and sends ACK #A2.
        ld      a, 9
        ld      (neo_fw_timeout_phase), a
        call    gs_boot_receive_data_byte
        jp      c, .upload_failed
        cp      NGS_FIRMWARE_READY_ACK
        jp      nz, .upload_failed

        xor     a
        ret

.upload_failed:
        scf
        ret

; -------------------------------------------------------------------------
; Standard GS105a ROM transport primitives.
; Commands use CBIT; data bytes use DBIT. All waits are bounded.
; -------------------------------------------------------------------------

; Probe stock ROM by reading the three-byte RAM size returned by command #20.
gs_boot_probe_rom:
        ld      a, GS_CMD_GET_TOTAL_RAM
        call    gs_boot_send_command_long
        ret     c
        call    gs_boot_receive_data_byte
        ret     c
        ld      (neo_fw_rom_ram_low), a
        call    gs_boot_receive_data_byte
        ret     c
        ld      (neo_fw_rom_ram_mid), a
        call    gs_boot_receive_data_byte
        ret     c
        ld      (neo_fw_rom_ram_high), a
        ld      a, (neo_fw_rom_ram_low)
        ld      b, a
        ld      a, (neo_fw_rom_ram_mid)
        or      b
        ld      b, a
        ld      a, (neo_fw_rom_ram_high)
        or      b
        jr      nz, .rom_probe_ok
        scf
        ret
.rom_probe_ok:
        or      a
        ret

; Long command wait for #F4 cold restart and ROM self-test.
gs_boot_send_command_long:
        ld      (neo_fw_command_byte), a
        ld      a, (neo_fw_command_byte)
        out     (GSCOM), a
        push    de
        ld      d, 16
.long_outer:
        ld      bc, FW_BYTE_TIMEOUT
.long_inner:
        in      a, (GSSTAT)
        ld      (neo_fw_last_status), a
        bit     B_CBIT, a
        jr      z, .long_done
        dec     bc
        ld      a, b
        or      c
        jr      nz, .long_inner
        dec     d
        jr      nz, .long_outer
        pop     de
        scf
        ret
.long_done:
        pop     de
        or      a
        ret

gs_boot_send_command:
        ld      (neo_fw_command_byte), a
        ld      a, (neo_fw_command_byte)
        out     (GSCOM), a
        ld      bc, FW_BYTE_TIMEOUT
.command_wait:
        in      a, (GSSTAT)
        ld      (neo_fw_last_status), a
        bit     B_CBIT, a
        jr      z, .command_done
        dec     bc
        ld      a, b
        or      c
        jr      nz, .command_wait
        scf
        ret
.command_done:
        or      a
        ret

; Preload the first argument byte of GS105a commands #13/#14. Those
; handlers read this byte before clearing CBIT; later arguments use DBIT.
gs_boot_write_command_argument:
        ld      (neo_fw_tx_byte), a
        ld      bc, FW_BYTE_TIMEOUT
.argument_wait_free:
        in      a, (GSSTAT)
        ld      (neo_fw_last_status), a
        bit     B_DBIT, a
        jr      z, .argument_slot_free
        dec     bc
        ld      a, b
        or      c
        jr      nz, .argument_wait_free
        scf
        ret
.argument_slot_free:
        ld      a, (neo_fw_tx_byte)
        out     (GSDAT), a
        or      a
        ret

gs_boot_send_data_byte:
        ld      (neo_fw_tx_byte), a
        ld      bc, FW_BYTE_TIMEOUT
.data_wait_free:
        in      a, (GSSTAT)
        ld      (neo_fw_last_status), a
        bit     B_DBIT, a
        jr      z, .data_slot_free
        dec     bc
        ld      a, b
        or      c
        jr      nz, .data_wait_free
        scf
        ret
.data_slot_free:
        ld      a, (neo_fw_tx_byte)
        out     (GSDAT), a
        ld      bc, FW_BYTE_TIMEOUT
.data_wait_consumed:
        in      a, (GSSTAT)
        ld      (neo_fw_last_status), a
        bit     B_DBIT, a
        jr      z, .data_consumed
        dec     bc
        ld      a, b
        or      c
        jr      nz, .data_wait_consumed
        scf
        ret
.data_consumed:
        push    hl
        ld      hl, (neo_fw_bytes_sent)
        inc     hl
        ld      (neo_fw_bytes_sent), hl
        pop     hl
        or      a
        ret

gs_boot_receive_data_byte:
        ld      bc, FW_BYTE_TIMEOUT
.receive_wait:
        in      a, (GSSTAT)
        ld      (neo_fw_last_status), a
        bit     B_DBIT, a
        jr      nz, .receive_ready
        dec     bc
        ld      a, b
        or      c
        jr      nz, .receive_wait
        scf
        ret
.receive_ready:
        in      a, (GSDAT)
        ld      (neo_fw_last_rx_byte), a
        or      a
        ret

; Project 124 private-protocol waits (custom firmware already running).
; -------------------------------------------------------------------------
neogs_wait_command_done::
        ld      bc, IPC_CMD_TIMEOUT
.cmd_wait:
        in      a, (GSSTAT)
        bit     B_CBIT, a
        jr      z, .cmd_done
        dec     bc
        ld      a, b
        or      c
        jr      nz, .cmd_wait
        scf
        ret
.cmd_done:
        or      a
        ret

neogs_wait_command_done_long::
        push    de
        ld      d, #10
.cmd_long_outer:
        ld      bc, IPC_CMD_TIMEOUT
.cmd_long_inner:
        in      a, (GSSTAT)
        bit     B_CBIT, a
        jr      z, .cmd_long_done
        dec     bc
        ld      a, b
        or      c
        jr      nz, .cmd_long_inner
        dec     d
        jr      nz, .cmd_long_outer
        pop     de
        scf
        ret
.cmd_long_done:
        pop     de
        or      a
        ret

neogs_wait_data_byte::
        ld      bc, IPC_BYTE_TIMEOUT
.data_wait:
        in      a, (GSSTAT)
        bit     B_DBIT, a
        jr      nz, .data_ready
        dec     bc
        ld      a, b
        or      c
        jr      nz, .data_wait
        scf
        ret

        or      a
        ret

; =========================================================================
; FILE: zx_main/x86_loader.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — TR-DOS task loader + NeoGS DMA / IPC
; Hardware protocol is based on NeoGS Programmer's Guide v0.1.A and
; ports.inc v0.3. The application-level command IDs/frame are Project 124.
;
; DMA mapping used here:
;   NeoGS physical DMA destination = $00D000
;   NeoGS Z80 logical DMA frame address = $5000
;   two-byte length header at $5000; x86 execution begins at $5002.
; This follows the guide's default mapping: logical $4000-$7FFF is the
; second half of physical 32K page 1 ($00C000-$00FFFF).
;
; Host DMA writes to Spectrum address $0000-$3FFF. For DMA WRITE there is
; NO dummy write. The dummy access described by the user is only for DMA READ.
; =========================================================================

; neogs_io.inc is included once by neogs_driver.asm before this module.

DMA_HOST_WINDOW    EQU #0000
DMA_WINDOW_LIMIT   EQU #3000

EXECUTE_X86_FILE_STAGE::
        ld      (P124_X86_REQUEST_FILENAME), hl
        ld      (P124_X86_REQUEST_TYPE), a
        ld      (P124_X86_LOADER_SP_BACKUP), sp
        ld      sp, #BFFF              ; buffer starts at #C000; stack below it

        MACRO_EF_STRING P124_msg_dbg_stack_isolated

        ; -------------------------------------------------------------
        ; Stage 5.0: deterministic diagnostic before accessing TR-DOS.
        ; B8 AA BB 40 90 F4 = MOV AX,BBAA / INC AX / NOP / HLT.
        ; Expected result: status=0, AX=BBAB.
        ; -------------------------------------------------------------
        MACRO_EF_STRING P124_msg_ipc_diag_start

        ; Exercise unknown-command release and explicit firmware reset before DMA.
        ld      a, #FE
        out     (GSCOM), a
        call    neogs_wait_command_done
        jp      c, .diag_protocol_error

        ld      a, CMD_RESET
        out     (GSCOM), a
        call    neogs_wait_command_done
        jp      c, .diag_protocol_error

        ld      hl, x86_diag_program
        ld      bc, x86_diag_program_end - x86_diag_program
        call    neogs_send_dma_block
        jp      c, .diag_dma_error

        call    neogs_run_and_get_result
        jp      c, .diag_protocol_error

        ld      a, (P124_X86_RESULT_STATUS)
        or      a
        jp      nz, .diag_execution_error
        ld      hl, (P124_X86_RESULT_AX)
        ld      de, #BBAB
        or      a
        sbc     hl, de
        jp      nz, .diag_wrong_result

        MACRO_EF_STRING P124_msg_ipc_diag_pass
        ld      hl, (P124_X86_RESULT_AX)
        MACRO_EF_HEX16 hl
        MACRO_EF_CHAR 13

        ; -------------------------------------------------------------
        ; Extended interpreter/transport self-test.
        ; Each case is sent through the real CBIT -> DMA -> DBIT path.
        ; Expected non-zero interpreter statuses are deliberate test results.
        ; -------------------------------------------------------------
        MACRO_EF_STRING P124_msg_ipc_edge_start

        ld      hl, x86_diag_inc_wrap
        ld      bc, x86_diag_inc_wrap_end - x86_diag_inc_wrap
        xor     a
        ld      de, #0000
        call    neogs_run_diag_case
        jp      c, .diag_extended_fail

        ld      hl, x86_diag_nop_hlt
        ld      bc, x86_diag_nop_hlt_end - x86_diag_nop_hlt
        xor     a
        ld      de, #0000
        call    neogs_run_diag_case
        jp      c, .diag_extended_fail

        ld      hl, x86_diag_unsupported
        ld      bc, x86_diag_unsupported_end - x86_diag_unsupported
        ld      a, 2
        ld      de, #0000
        call    neogs_run_diag_case
        jp      c, .diag_extended_fail

        ld      hl, x86_diag_truncated_mov
        ld      bc, x86_diag_truncated_mov_end - x86_diag_truncated_mov
        ld      a, 4
        ld      de, #0000
        call    neogs_run_diag_case
        jp      c, .diag_extended_fail

        ld      hl, x86_diag_fallthrough
        ld      bc, x86_diag_fallthrough_end - x86_diag_fallthrough
        ld      a, 4
        ld      de, #0000
        call    neogs_run_diag_case
        jp      c, .diag_extended_fail

        MACRO_EF_STRING P124_msg_ipc_edge_pass

        ; -------------------------------------------------------------
        ; NeoGS x86 memory test: 64 pages plus word/segment addressing cases.
        ; Destructive only to the reserved x86 memory region.
        ; -------------------------------------------------------------
        MACRO_EF_STRING P124_msg_memtest_start
        call    neogs_mem_test_and_get_result
        jp      c, .diag_extended_fail
        ld      a, (P124_X86_RESULT_STATUS)
        or      a
        jp      nz, .diag_memtest_fail
        ld      hl, (P124_X86_RESULT_AX)
        ld      de, 1
        or      a
        sbc     hl, de
        jp      nz, .diag_memtest_fail
        MACRO_EF_STRING P124_msg_memtest_pass

        ; BIOS foundation POST. Uses the same command/result channel; DMA is untouched.
        MACRO_EF_STRING P124_msg_bios_post_start
        call    neogs_bios_post_and_get_result
        jp      c, .bios_post_transport_fail
        ld      a, (P124_X86_RESULT_STATUS)
        or      a
        jp      nz, .bios_post_fail
        ld      hl, (P124_X86_RESULT_AX)
        ld      de, #B105
        or      a
        sbc     hl, de
        jp      nz, .bios_post_fail
        MACRO_EF_STRING P124_msg_bios_post_pass

        ; -------------------------------------------------------------
        ; Stage 2A: 8086 real-mode INT/IRET frame and stack test.
        ; This validates IVT lookup, FLAGS/CS/IP push order, IF/TF clearing,
        ; IRET restoration, and SP balance. Guest handler execution is Stage 2B.
        ; -------------------------------------------------------------
        MACRO_EF_STRING P124_msg_int_test_start
        call    neogs_int_test_and_get_result
        jp      c, .int_test_transport_fail
        ld      a, (P124_X86_RESULT_STATUS)
        or      a
        jp      nz, .int_test_fail
        ld      hl, (P124_X86_RESULT_AX)
        ld      de, #1A01
        or      a
        sbc     hl, de
        jp      nz, .int_test_fail
        MACRO_EF_STRING P124_msg_int_test_pass

        ; -------------------------------------------------------------
        ; Stage 2B: execute a guest INT 20h handler through CS:IP, then IRET.
        ; The guest handler changes AX and returns to the caller's HLT.
        ; -------------------------------------------------------------
        MACRO_EF_STRING P124_msg_int_guest_start
        call    neogs_int_guest_test_and_get_result
        jp      c, .int_guest_transport_fail
        ld      a, (P124_X86_RESULT_STATUS)
        or      a
        jp      nz, .int_guest_fail
        ld      hl, (P124_X86_RESULT_AX)
        ld      de, #1A02
        or      a
        sbc     hl, de
        jp      nz, .int_guest_fail
        MACRO_EF_STRING P124_msg_int_guest_pass

        ; -------------------------------------------------------------
        ; Stage 3A: guest INT 10h video BIOS services and live console stream.
        ; The NeoGS sends each teletype byte over DBIT while this host drains
        ; the stream to port #EF and the 42x12 6x8 debug window.
        ; -------------------------------------------------------------
        MACRO_EF_STRING P124_msg_bios_video_start
        call    neogs_bios_video_test_and_get_result
        jp      c, .bios_video_transport_fail
        ld      a, (P124_X86_RESULT_STATUS)
        or      a
        jp      nz, .bios_video_fail
        ld      hl, (P124_X86_RESULT_AX)
        ld      de, #1A10
        or      a
        sbc     hl, de
        jp      nz, .bios_video_fail
        MACRO_EF_STRING P124_msg_bios_video_pass

        ; -------------------------------------------------------------
        ; Stage 5A: search TR-DOS directory entry.
        ; Name #5CDD..#5CE4, type #5CE5, per the working project path.
        ; -------------------------------------------------------------
        ld      hl, (P124_X86_REQUEST_FILENAME)
        ld      de, #5CDD
        ld      bc, 8
        ldir
        ld      a, (P124_X86_REQUEST_TYPE)
        ld      (#5CE5), a
        ld      a, 9
        ld      (#5D06), a
        ld      c, 10
        call    #3D13
        ld      a, c
        cp      #FF
        jp      z, .file_not_found

        ; C=#0A returns only the catalogue entry number. Fetch the full
        ; 16-byte descriptor explicitly with C=#08 before using #5CE8..#5CEC.
        ld      a, c
        push    af
        xor     a
        ld      (#5D0F), a
        ld      (#5D10), a
        pop     af
        ld      c, 8
        call    #3D13
        ld      a, (#5D0F)
        ld      b, a
        ld      a, (#5D10)
        or      b
        jp      nz, .bad_file_descriptor

        MACRO_EF_STRING P124_msg_file_ok

        ; -------------------------------------------------------------
        ; Stage 5B: validate descriptor BEFORE reading any sector.
        ; #5CE8 = byte length, #5CEA = sector count,
        ; #5CEB = first internal sector (0..15), #5CEC = track.
        ; The DMA frame window is $5000..$7FFF (12 KiB). Its first two
        ; bytes are the length header, so executable payload max is $2FFE.
        ; -------------------------------------------------------------
        ld      bc, (#5CE8)
        ld      a, b
        or      c
        jp      z, .empty_file_error
        ld      (P124_X86_FILE_LENGTH), bc

        ; Reject files larger than the executable 12 KB window minus header.
        ld      a, b
        cp      #2F
        jr      c, .file_length_ok
        jp      nz, .task_too_large
        ld      a, c
        cp      #FF
        jp      nc, .task_too_large
.file_length_ok:
        ; Sector count must equal ceil(file_length / 256).
        ld      hl, (P124_X86_FILE_LENGTH)
        ld      de, 255
        add     hl, de
        ld      e, h
        ld      a, (#5CEA)
        or      a
        jp      z, .empty_file_error
        cp      e
        jp      nz, .bad_file_descriptor
        ld      (P124_X86_READ_REMAINING), a

        ld      a, (#5CEB)
        cp      16
        jp      nc, .bad_file_descriptor
        ld      (P124_X86_READ_SECTOR), a
        ld      a, (#5CEC)
        ld      (P124_X86_READ_TRACK), a
        ld      hl, TRDOS_PANEL_BUFFER
        ld      (P124_X86_READ_BUFFER), hl

.read_sector_loop:
        ; Read exactly one sector per TR-DOS #05 call.
        ld      a, (P124_X86_READ_TRACK)
        ld      d, a
        ld      a, (P124_X86_READ_SECTOR)
        ld      e, a
        ld      hl, (P124_X86_READ_BUFFER)
        ld      b, 1
        ld      c, 5
        xor     a
        ld      (#5D0F), a
        ld      (#5D10), a
        call    #3D13
        ; C=#05 stores its two-byte error result at #5D0F/#5D10.
        ld      a, (#5D0F)
        ld      b, a
        ld      a, (#5D10)
        or      b
        jp      nz, .sector_read_error

        ; Advance host buffer by 256 bytes.
        ld      hl, (P124_X86_READ_BUFFER)
        ld      de, 256
        add     hl, de
        ld      (P124_X86_READ_BUFFER), hl

        ; TR-DOS sectors are numbered 0..15 in this project convention.
        ld      a, (P124_X86_READ_SECTOR)
        inc     a
        cp      16
        jr      c, .sector_same_track
        xor     a
        ld      (P124_X86_READ_SECTOR), a
        ld      a, (P124_X86_READ_TRACK)
        inc     a
        ld      (P124_X86_READ_TRACK), a
        jr      .sector_advanced

.sector_same_track:
        ld      (P124_X86_READ_SECTOR), a

.sector_advanced:
        ld      a, (P124_X86_READ_REMAINING)
        dec     a
        ld      (P124_X86_READ_REMAINING), a
        jp      nz, .read_sector_loop

        MACRO_EF_STRING P124_msg_buf_loaded

        ; -------------------------------------------------------------
        ; Stage 5C: transfer actual task. The DMA frame contains a two-byte
        ; length prefix, therefore the total transfer must fit in $3000 bytes.
        ; -------------------------------------------------------------
        ld      bc, (P124_X86_FILE_LENGTH)
        ld      hl, TRDOS_PANEL_BUFFER
        call    neogs_send_dma_block
        jp      c, .task_dma_error
        MACRO_EF_STRING P124_msg_task_dma_ok

        ; -------------------------------------------------------------
        ; Stage 5D: execute and fetch framed result.
        ; -------------------------------------------------------------
        call    neogs_run_x86_real_and_get_result
        jp      c, .result_timeout

        MACRO_EF_STRING P124_msg_x86_result_ready
        ld      a, (P124_X86_RESULT_STATUS)
        ld      l, a
        ld      h, 0
        MACRO_EF_HEX16 hl
        MACRO_EF_CHAR 32
        ld      hl, (P124_X86_RESULT_AX)
        MACRO_EF_HEX16 hl
        MACRO_EF_CHAR 13

        ld      a, (P124_X86_RESULT_STATUS)
        or      a
        jp      nz, .interpreter_error

        ld      hl, (P124_X86_RESULT_AX)
        call    BIOS_DebugConsole_PrintHex16
        ld      a, 13
        call    BIOS_DebugConsole_PrintChar
        ld      a, 10
        call    BIOS_DebugConsole_PrintChar
        or      a
        jp      .loader_restore_stack

; -------------------------------------------------------------------------

; =========================================================================
; NeoGS transport helpers — exact CBIT/DBIT meanings from Guide v0.1.A
; =========================================================================

; IN: HL=source in host RAM, BC=payload byte count (1..#2FFE).
; OUT: Carry clear after NeoGS armed DMA, consumed the host end marker,
;      and disabled DMA. Carry set on timeout.
;
; Protocol:
;   host OUT GSCOM,CMD_LOAD_X86 -> CBIT=1
;   NeoGS configures DMA, clears CLRCBIT -> CBIT=0 (READY)
;   host LDIR to #0000..#3FFF (no dummy write)
;   host OUT GSCOM,#FF -> CBIT=1 (end marker)
;   NeoGS disables DMA, clears CLRCBIT -> CBIT=0 (DONE)
neogs_send_dma_block:
        ld      a, b
        or      c
        jp      z, .dma_bad_length
        ld      a, b
        cp      #40
        jr      c, .dma_length_ok
        jp      nz, .dma_bad_length
        ld      a, c
        or      a
        jp      nz, .dma_bad_length
.dma_length_ok:
        ld      (P124_DMA_SOURCE), hl
        ld      (P124_DMA_LENGTH), bc

        ; Validate payload + 2-byte length prefix <= $3000.
        push    bc
        pop     hl
        ld      de, 2
        add     hl, de
        ld      a, h
        cp      #30
        jr      c, .dma_total_length_ok
        jp      nz, .dma_bad_length
        ld      a, l
        or      a
        jp      nz, .dma_bad_length
.dma_total_length_ok:
        ld      a, CMD_LOAD_X86
        out     (GSCOM), a
        call    neogs_wait_command_done
        ret     c

        ; LDIR is used for BOTH pieces of the frame. This is intentional:
        ; the DMA engine observes host memory-write cycles. The first LDIR
        ; writes the two-byte header to host addresses #0000-#0001; the second
        ; writes the payload to #0002 onward. No direct LD (nn),A writes and
        ; no temporary stack are needed.
        di
        ld      hl, (P124_DMA_LENGTH)
        ld      (P124_DMA_HEADER), hl
        ld      hl, P124_DMA_HEADER
        ld      de, DMA_HOST_WINDOW
        ld      bc, 2
        ldir

        ld      hl, (P124_DMA_SOURCE)
        ld      de, DMA_HOST_WINDOW + 2
        ld      bc, (P124_DMA_LENGTH)
        ldir

        ; End marker is a command-byte write; it is NOT a DMA data byte.
        ld      a, #FF
        out     (GSCOM), a
        call    neogs_wait_command_done
        ret

.dma_bad_length:
        scf
        ret

; Wait for the RUN command while servicing the bidirectional data channel.
; Stream protocol: nonzero=character, 00 00=clear, 00 01 row col=set cursor.
; Each character is sent to emulator printer port #EF and rendered on screen.
neogs_wait_run_done_with_console:
        push    de
        ld      d, #10
.run_outer:
        ld      hl, IPC_CMD_TIMEOUT
.run_inner:
        in      a, (GSSTAT)
        bit     B_DBIT, a
        jr      z, .no_stream_byte
        in      a, (GSDAT)
        or      a
        jr      z, .stream_control
        push    af
        ld      bc, #00EF
        out     (c), a
        pop     af
        call    BIOS_DebugConsole_PrintChar
        jr      .run_inner

.stream_control:
        call    neogs_wait_data_byte
        jr      c, .run_timeout
        or      a
        jr      z, .stream_clear
        cp      1
        jr      nz, .run_inner
        call    neogs_wait_data_byte
        jr      c, .run_timeout
        ld      (P124_BIOS_STREAM_ROW), a
        call    neogs_wait_data_byte
        jr      c, .run_timeout
        ld      c, a
        ld      a, (P124_BIOS_STREAM_ROW)
        call    BIOS_DebugConsole_SetCursor
        jr      .run_inner

.stream_clear:
        call    BIOS_DebugConsole_Clear
        jr      .run_inner

.no_stream_byte:
        bit     B_CBIT, a
        jr      z, .run_done
        dec     hl
        ld      a, h
        or      l
        jr      nz, .run_inner
        dec     d
        jr      nz, .run_outer
.run_timeout:
        pop     de
        scf
        ret
.run_done:
        pop     de
        or      a
        ret

        out     (GSCOM), a

        call    neogs_wait_data_byte
        ret     c

        ret     c

        ret

; ИНТЕЛЛЕКТУАЛЬНАЯ БЕЗОПАСНАЯ НАСТРОЙКА CMOS ЧАСОВ
; =========================================================================
CHECK_AND_INIT_CMOS::                   
        PUSH    AF: PUSH BC
        CALL    open_cmos
        
        ; Проверяем сигнатуру наличия часов в ячейке #11
        LD      A, #11
        CALL    read_byte_raw
        CP      #AA
        JR      Z, .already_active      
        
        ; Холодный старт: записываем обязательные системные константы
        LD      A, #11
        LD      L, #AA
        CALL    write_byte_raw         ; Маркер готовности системы
        
        LD      A, #0A
        LD      L, #20
        CALL    write_byte_raw         ; Регистр А: Включение кварца 32кГц
        
        LD      A, #0B
        LD      L, #02
        CALL    write_byte_raw         ; Регистр B: 24-часовой BCD формат

.already_active
        CALL    close_cmos
        POP     BC: POP     AF
        RET

; =========================================================================
; ПРОЦЕДУРА ЧТЕНИЯ ТЕКУЩЕГО ВРЕМЕНИ ИЗ ЧИПА RTC
; =========================================================================
READ_CURRENT_TIME::                     
        PUSH    AF: PUSH BC: PUSH HL
        DI
        CALL    open_cmos

.wait_rtc
        ; Контролируем флаг обновления UIP (битовая маска #80 в регистре #0A)
        LD      A, #0A
        CALL    read_byte_raw
        RLA                             
        JR      C, .wait_rtc            

        ; Вычитываем BCD данные с атомарным сохранением в текстовые буферы
        ; 1. Секунды (Ячейка #00)
        LD      A, #00
        CALL    read_byte_raw
        CALL    split_bcd
        LD      A, H: LD (txt_sec+0), A
        LD      A, L: LD (txt_sec+1), A

        ; 2. Минуты (Ячейка #02)
        LD      A, #02
        CALL    read_byte_raw
        CALL    split_bcd
        LD      A, H: LD (txt_min+0), A
        LD      A, L: LD (txt_min+1), A

        ; 3. Часы (Ячейка #04)
        LD      A, #04
        CALL    read_byte_raw
        CALL    split_bcd
        LD      A, H: LD (txt_hour+0), A
        LD      A, L: LD (txt_hour+1), A

        CALL    close_cmos
        EI
        POP     HL: POP BC: POP AF
        RET

; --- Атомарное низкоуровневое чтение байта из К512ВИ1 ---
read_byte_raw
        LD      BC, #DFF7
        OUT     (C), A                  
        LD      BC, #BFF7
        IN      A, (C)                  
        RET

; --- Атомарная низкоуровневая запись байта в К512ВИ1 ---
write_byte_raw
        LD      BC, #DFF7
        OUT     (C), A
        LD      BC, #BFF7
        OUT     (C), L
        RET

open_cmos
        LD      BC, #EFF7
        LD      A, #80
        OUT     (C), A
        RET

close_cmos
        LD      BC, #EFF7
        XOR     A
        OUT     (C), A
        RET

; --- Распаковка BCD полубайтов в ASCII символы ---
split_bcd
        PUSH    AF
        AND     %11110000               
        RRCA: RRCA: RRCA: RRCA
        ADD     A, 48
        LD      H, A
        POP     AF
        AND     %00001111               
        ADD     A, 48
        LD      L, A
        RET

LOG_WriteMessageToBuffer::
        PUSH    AF: PUSH BC: PUSH DE: PUSH HL: PUSH IX
        PUSH    HL: POP IX                      

.loop_text:
        LD      A, (IX+0): OR A: JR Z, .msg_finished
        LD      B, A                    

        ; Отправляем символ напрямую в физический знакоместный рендерер экрана
        CALL    Direct_PrintChar           
        INC     IX: JR .loop_text

.msg_finished:
        ; Автоматически закрываем строку лога переводом каретки CR/LF
        CALL    LOG_CloseActiveLine

        POP     IX: POP HL: POP DE: POP BC: POP AF
        RET

LOG_CloseActiveLine::
        PUSH    AF: PUSH BC: PUSH DE: PUSH HL
        LD      A, #0D: CALL Direct_PrintChar
        LD      A, #0A: CALL Direct_PrintChar
        POP     HL: POP DE: POP BC: POP AF
        RET

Util_PrintHex16Screen::

        LD      A, L

; -------------------------------------------------------------------------
; ВЫВОД 8-БИТНОГО HEX-ЗНАЧЕНИЯ ИЗ L НА ЭКРАН ХОСТА

Util_PrintHex8Screen::

        PUSH    BC

        POP     AF

; -------------------------------------------------------------------------
; ГЛОБАЛЬНАЯ ПОДПРОГРАММА ПОБАЙТОВОГО РЕНДЕРА И ВЫВОДА В HEX НА ЭКРАН
; -------------------------------------------------------------------------
Util_Internal_ScreenHex8::
        PUSH    AF
        RRA: RRA: RRA: RRA                 ; Сдвигаем старший ниббл вниз
        CALL    .local_nibble
        POP     AF
.local_nibble:
        AND     #0F                        ; Изолируем 4 бита
        CP      10
        JR      C, .is_digit
        ADD     A, 7                       ; Коррекция для символов A..F
.is_digit:
        ADD     A, "0"                     ; Перевод в ASCII-код
        CALL    Direct_PrintChar           ; Печать символа в знакоместо экрана
        RET

; =========================================================================
; =========================================================================
; LOWER SCREEN BITMAP HEX DUMP — PORT #EF
; Dumps ONLY the Spectrum bitmap for the lower half: Y=96..191, 3072 bytes.
; Color attributes are deliberately excluded.
; Output is ASCII HEX with scanline labels and Spectrum bitmap addresses.
; IN: A = frame number.
; =========================================================================
P124_DumpLowerScreen::
        push    af
        push    bc
        push    de
        push    hl
        ld      (P124_VRAM_FRAME), a
        ld      hl, P124_VRAM_DUMP_HEADER
        call    P124_VRAM_DumpString
        ld      a, (P124_VRAM_FRAME)
        call    P124_VRAM_DumpHex8
        ld      hl, P124_VRAM_BITMAP_HEADER
        call    P124_VRAM_DumpString

        ld      a, 96
        ld      (P124_VRAM_Y), a
.bitmap_line:
        ld      a, (P124_VRAM_Y)
        ld      c, 0
        call    bios_dbg_calc_addr
        ld      (P124_VRAM_PTR), hl

        ld      hl, P124_VRAM_Y_LABEL
        call    P124_VRAM_DumpString
        ld      a, (P124_VRAM_Y)
        call    P124_VRAM_DumpHex8
        ld      a, ' '
        call    P124_VRAM_DumpPutChar
        ld      a, '@'
        call    P124_VRAM_DumpPutChar
        ld      hl, (P124_VRAM_PTR)
        call    P124_VRAM_DumpHex16
        ld      a, ':'
        call    P124_VRAM_DumpPutChar
        ld      a, ' '
        call    P124_VRAM_DumpPutChar

        ld      b, 32
.bitmap_byte:
        ld      hl, (P124_VRAM_PTR)
        ld      a, (hl)
        call    P124_VRAM_DumpHex8
        ld      hl, (P124_VRAM_PTR)
        inc     hl
        ld      (P124_VRAM_PTR), hl
        ld      a, ' '
        call    P124_VRAM_DumpPutChar
        djnz    .bitmap_byte

        call    P124_VRAM_DumpCRLF
        ld      a, (P124_VRAM_Y)
        inc     a
        ld      (P124_VRAM_Y), a
        cp      192
        jr      c, .bitmap_line

        ld      hl, P124_VRAM_DUMP_END
        call    P124_VRAM_DumpString
        pop     hl
        pop     de
        pop     bc
        pop     af
        ret

P124_VRAM_DumpString:
        push    hl
.string_next:
        ld      a, (hl)
        or      a
        jr      z, .string_done
        call    P124_VRAM_DumpPutChar
        inc     hl
        jr      .string_next
.string_done:
        pop     hl
        ret

; A = byte, output two uppercase hexadecimal digits.
P124_VRAM_DumpHex8:
        push    af
        rrca
        rrca
        rrca
        rrca
        call    P124_VRAM_DumpNibble
        pop     af
        call    P124_VRAM_DumpNibble
        ret

; HL = 16-bit value, output four uppercase hexadecimal digits.
P124_VRAM_DumpHex16:
        push    hl
        ld      a, h
        call    P124_VRAM_DumpHex8
        ld      a, l
        call    P124_VRAM_DumpHex8
        pop     hl
        ret

P124_VRAM_DumpNibble:
        and     #0F
        cp      10
        jr      c, .nibble_digit
        add     a, 7
.nibble_digit:
        add     a, '0'
        jp      P124_VRAM_DumpPutChar

; A = ASCII character. Preserve caller registers used by dump loops.
P124_VRAM_DumpPutChar:
        push    af
        push    bc
        ld      bc, #00EF
        out     (c), a
        pop     bc
        pop     af
        ret

P124_VRAM_DumpCRLF:
        push    af
        ld      a, 13
        call    P124_VRAM_DumpPutChar
        ld      a, 10
        call    P124_VRAM_DumpPutChar
        pop     af
        ret

P124_VRAM_DUMP_HEADER:   defb 13,10,"===== P124 BITMAP FRAME ",0
P124_VRAM_BITMAP_HEADER: defb " LOWER SCREEN PIXELS Y=96..191 (32 BYTES/SCANLINE) =====",13,10,0
P124_VRAM_Y_LABEL:       defb "Y=",0
P124_VRAM_DUMP_END:      defb "===== END P124 BITMAP FRAME =====",13,10,0
P124_VRAM_FRAME:         db 0
P124_VRAM_Y:             db 0
P124_VRAM_PTR:           dw 0

; --- ПРОЦЕДУРА ПРЯМОЙ ПЕЧАТИ НИЖНЕГО ОТЛАДЧИКА С АВТО-СКРОЛЛОМ ---
Direct_PrintChar::
        PUSH    BC
        PUSH    DE
        PUSH    HL
        LD      B, A                    

        ; Тотальный аппаратный перехват в физический порт отладки #EF
        IF ENABLE_HARDWARE_DEBUG == 1
        PUSH    BC
        LD      BC, #00EF
        OUT     (C), A                  
        POP     BC
        ENDIF

        CP      #0D                     ; Проверяем возврат каретки (CR)
        JR      NZ, .check_lf
        XOR     A
        LD      (Console_X), A          
        JP      .done_print             

.check_lf:
        CP      #0A                     ; Проверяем перевод строки (LF)
        JR      NZ, .render_gate
        XOR     A
        LD      (Console_X), A          
        LD      A, (Console_Y)
        INC     A                       
        LD      (Console_Y), A
        CALL    EnsureConsoleSpace      
        JP      .done_print             

.render_gate:
        LD      A, (Console_Y)
        LD      D, A                    
        LD      A, (Console_X)
        LD      C, A                    
        
        LD      A, B                    
        CALL    Direct_PrintChar_Args   

        LD      A, (Console_X)
        INC     A
        LD      (Console_X), A          

        CP      32                      ; Автоперенос по правому краю
        JR      C, .done_print          
        
        XOR     A
        LD      (Console_X), A          
        LD      A, (Console_Y)
        INC     A                       
        LD      (Console_Y), A
        CALL    EnsureConsoleSpace      

.done_print:
        POP     HL
        POP     DE
        POP     BC
        RET

; --- ТОЧНЫЙ КAНОНИЧЕСКИЙ РЕНДЕРЕР ЗНAКОМЕСТА ИЗ ПРОЕКТА 123 ---
Direct_PrintChar_Args::
        PUSH    AF
        PUSH    BC
        PUSH    DE
        PUSH    HL

        ; 1. РАСЧЕТ ТОЧНОГО АДРЕСА АТРИБУТА ЦВЕТА (#5800)
        LD      A, D                    ; Console_Y (0..23)
        AND     #18                     
        RRCA: RRCA: RRCA
        ADD     A, #58                  
        LD      H, A
        
        LD      A, D
        AND     #07
        RLCA: RLCA: RLCA: RLCA: RLCA    
        LD      L, A
        
        LD      A, C                    ; Console_X (0..31)
        ADD     A, L
        LD      L, A                    

        LD      A, D                    
        CP      12                      
        JR      NC, .set_black_bg
        LD      A, %01001111            ; Верх (0..11): PAPER=1(Синий), INK=7(Белый), BRIGHT=1
        JR      .write_color
.set_black_bg:
        LD      A, %01000100            ; Низ (12..23): PAPER=0(Черный), INK=4(Зеленый), BRIGHT=1
.write_color:
        LD      (HL), A                 

        ; 2. КAНОНИЧЕСКИЙ РАСЧЕТ АДРЕСА ПИКСЕЛЕЙ В ЭКРАНЕ (#4000)
        LD      A, D                    
        AND     #18
        ADD     A, #40                  
        LD      H, A                    
        
        LD      A, D                    
        AND     #07
        RRCA: RRCA: RRCA
        LD      L, A
        
        LD      A, L
        AND     #E0
        ADD     A, C                    
        LD      E, A
        
        LD      A, L
        AND     #07
        OR      H
        LD      D, A                    ; DE = базовый адрес пикселей на экране

        ; 3. ПОСТРОЧНЫЙ РЕНДЕР ШРИФТА ИЗ ВЫБРАННОГО МАССИВА ОБОЛОЧКИ
        LD      A, B                    
        SUB     32
        LD      L, A
        LD      H, 0
        ADD     HL, HL
        ADD     HL, HL
        ADD     HL, HL                  ; HL = смещение буквы (ASCII * 8)
        
        PUSH    DE
        LD      A, (UI_Active_Font)
        AND     A
        JR      NZ, .low_font68
        LD      BC, FONT_DATA_88
        JR      .low_font_done
.low_font68:
        LD      BC, FONT_DATA_68
.low_font_done:
        ADD     HL, BC                  
        POP     DE

        ; Отрисовка 8 вертикальных линий
        LD      B, 8
.draw_font_loop:
        LD      A, (HL)
        LD      (DE), A
        INC     HL
        INC     D                       
        DJNZ    .draw_font_loop

        POP     HL
        POP     DE
        POP     BC
        POP     AF
        RET

; =========================================================================
; FILE: zx_main/bios_console.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — BIOS 6x8 debug console backend
; Lower Spectrum window: 42 columns x 12 rows, 6 pixels x 8 scanlines.
; Uses font68.bin (96 glyphs, 8 bytes per glyph). This renderer is separate
; from Direct_PrintChar so existing 32-column Commander panels are unchanged.
; =========================================================================

BIOS_DEBUG_COLUMNS EQU 42
BIOS_DEBUG_ROWS    EQU 12
BIOS_DEBUG_TOP     EQU 12

BIOS_DebugConsole_Init::
        ld      a, (Console_Y)
        sub     BIOS_DEBUG_TOP
        jr      nc, .row_ok
        xor     a
.row_ok:
        cp      BIOS_DEBUG_ROWS
        jr      c, .row_valid
        ld      a, BIOS_DEBUG_ROWS - 1
.row_valid:
        ld      (bios_dbg_row), a
        ld      a, (Console_X)
        cp      BIOS_DEBUG_COLUMNS
        jr      c, .col_valid
        xor     a
.col_valid:
        ld      (bios_dbg_col), a
        ret

; Clear only the lower 96-pixel-high window, not the Commander file panel.
BIOS_DebugConsole_Clear::
        push    af
        push    bc
        push    de
        push    hl
        ld      a, 96
        ld      (bios_dbg_clear_y), a
.clear_scanline:
        ld      a, (bios_dbg_clear_y)
        ld      c, 0
        call    bios_dbg_calc_addr
        push    hl
        pop     de
        inc     de
        ld      (hl), 0
        ld      bc, 31
        ldir
        ld      a, (bios_dbg_clear_y)
        inc     a
        ld      (bios_dbg_clear_y), a
        cp      192
        jr      c, .clear_scanline
        ld      hl, #5980
        ld      de, #5981
        ld      (hl), #44
        ld      bc, 383
        ldir
        xor     a
        ld      (bios_dbg_row), a
        ld      (bios_dbg_col), a
        ld      (Console_Y), a
        ld      a, BIOS_DEBUG_TOP
        ld      (Console_Y), a
        xor     a
        ld      (Console_X), a
        pop     hl
        pop     de
        pop     bc
        pop     af
        ret

; IN: A=row, C=column. Invalid coordinates are ignored.
BIOS_DebugConsole_SetCursor::
        cp      BIOS_DEBUG_ROWS
        ret     nc
        ld      (bios_dbg_row), a
        ld      a, c
        cp      BIOS_DEBUG_COLUMNS
        ret     nc
        ld      (bios_dbg_col), a
        ret

; IN: A=one output byte from the NeoGS stream.
; Printable bytes are drawn as packed 6-pixel glyphs; CR/LF/BS are handled.
BIOS_DebugConsole_PrintChar::
        push    af
        push    bc
        push    de
        push    hl
        ld      (bios_dbg_char), a

        cp      13
        jp      z, .carriage_return
        cp      10
        jp      z, .line_feed
        cp      8
        jp      z, .backspace
        cp      7
        jp      z, .done
        cp      32
        jp      c, .done
        cp      128
        jr      c, .glyph_range_ok
        ld      a, '?'
        ld      (bios_dbg_char), a
.glyph_range_ok:
        ld      a, (bios_dbg_char)
        sub     32
        ld      l, a
        ld      h, 0
        add     hl, hl
        add     hl, hl
        add     hl, hl
        ld      de, FONT_DATA_68
        add     hl, de
        ld      (bios_dbg_font_ptr), hl

        ; Eight scanlines share the same character-cell X origin.
        ; Reset the pixel coordinate for every glyph scanline.
        xor     a
        ld      (bios_dbg_scan), a

.glyph_scanline:
        ; Pixel X = column * 6. Recompute on EVERY scanline:
        ; glyph_pixel advances bios_dbg_xpixel by six pixels, so carrying
        ; that value into the next scanline causes diagonal glyph rendering.
        ld      a, (bios_dbg_col)
        add     a, a
        ld      c, a
        add     a, a
        add     a, c
        ld      (bios_dbg_xpixel), a

        ld      hl, (bios_dbg_font_ptr)
        ld      a, (hl)
        ld      (bios_dbg_bits), a
        inc     hl
        ld      (bios_dbg_font_ptr), hl

        ; Absolute Spectrum scanline = 96 + row*8 + scan.
        ld      a, (bios_dbg_row)
        add     a, a
        add     a, a
        add     a, a
        add     a, 96
        ld      b, a
        ld      a, (bios_dbg_scan)
        add     a, b
        ld      (bios_dbg_y), a

        ld      b, 6
.glyph_pixel:
        ld      a, (bios_dbg_xpixel)
        ld      c, a
        and     7
        ld      (bios_dbg_bitpos), a
        ld      a, c
        srl     a
        srl     a
        srl     a
        ld      c, a
        ld      a, (bios_dbg_y)
        call    bios_dbg_calc_addr

        ld      a, (bios_dbg_bitpos)
        ld      c, a
        ld      a, #80
.mask_loop:
        ld      d, a
        ld      a, c
        or      a
        ld      a, d
        jr      z, .mask_ready
        srl     a
        dec     c
        jr      .mask_loop
.mask_ready:
        ld      (bios_dbg_mask), a
        ld      a, (bios_dbg_bits)
        and     #20
        jr      z, .pixel_clear
        ld      a, (hl)
        ld      e, a
        ld      a, (bios_dbg_mask)
        or      e
        ld      (hl), a
        jr      .pixel_done
.pixel_clear:
        ld      a, (bios_dbg_mask)
        cpl
        ld      e, a
        ld      a, (hl)
        and     e
        ld      (hl), a
.pixel_done:
        ld      a, (bios_dbg_bits)
        add     a, a
        ld      (bios_dbg_bits), a
        ld      a, (bios_dbg_xpixel)
        inc     a
        ld      (bios_dbg_xpixel), a
        djnz    .glyph_pixel

        ld      a, (bios_dbg_scan)
        inc     a
        ld      (bios_dbg_scan), a
        cp      8
        jp      c, .glyph_scanline

        ; The full-screen glyph test suppresses only the final auto-wrap,
        ; so row 11 remains visible instead of scrolling the test's first row.
        ld      a, (bios_dbg_no_wrap)
        or      a
        jr      nz, .done
        ld      a, (bios_dbg_col)
        inc     a
        ld      (bios_dbg_col), a
        cp      BIOS_DEBUG_COLUMNS
        jr      c, .done
        xor     a
        ld      (bios_dbg_col), a
        call    bios_dbg_newline
        jr      .done

.carriage_return:
        xor     a
        ld      (bios_dbg_col), a
        jr      .done
.line_feed:
        call    bios_dbg_newline
        jr      .done
.backspace:
        ld      a, (bios_dbg_col)
        or      a
        jr      z, .done
        dec     a
        ld      (bios_dbg_col), a
.done:
        pop     hl
        pop     de
        pop     bc
        pop     af
        ret

; Full-screen renderer test: 12 rows, 42 identical glyphs per row.
; Rows A-L are drawn through the same 6x8 PrintChar routine used by INT 10h.
; The final glyph suppresses automatic wrap to prevent scrolling row A away.
BIOS_DebugConsole_TestPattern::
        call    BIOS_DebugConsole_Clear
        xor     a
        ld      (bios_dbg_test_row), a
        ld      a, 'A'
        ld      (bios_dbg_test_char), a

.test_row:
        ld      a, (bios_dbg_test_row)
        ld      c, 0
        call    BIOS_DebugConsole_SetCursor
        ld      a, (bios_dbg_test_row)
        cp      BIOS_DEBUG_ROWS - 1
        jr      nz, .row_started
        xor     a
        ld      (bios_dbg_test_col), a
        ld      b, BIOS_DEBUG_COLUMNS
        jr      .test_char
.row_started:
        xor     a
        ld      (bios_dbg_test_col), a
        ld      b, BIOS_DEBUG_COLUMNS

.test_char:
        ld      a, (bios_dbg_test_row)
        cp      BIOS_DEBUG_ROWS - 1
        jr      nz, .emit_char
        ld      a, (bios_dbg_test_col)
        cp      BIOS_DEBUG_COLUMNS - 1
        jr      nz, .emit_char
        ld      a, 1
        ld      (bios_dbg_no_wrap), a
.emit_char:
        ld      a, (bios_dbg_test_char)
        ; Preserve B: it is the 42-character loop counter.
        push    bc
        ld      bc, #00EF
        out     (c), a
        pop     bc
        call    BIOS_DebugConsole_PrintChar
        xor     a
        ld      (bios_dbg_no_wrap), a
        ld      a, (bios_dbg_test_col)
        inc     a
        ld      (bios_dbg_test_col), a

        ; Frame 1: capture the first three glyphs (AAA) on row 0.
        ld      a, (bios_dbg_test_col)
        cp      3
        jr      nz, .check_full_row_frame
        ld      a, 1
        call    P124_DumpLowerScreen

.check_full_row_frame:
        ; Frame 2: capture the complete 42-character first row.
        ld      a, (bios_dbg_test_col)
        cp      BIOS_DEBUG_COLUMNS
        jr      nz, .continue_test_row
        ld      a, 2
        call    P124_DumpLowerScreen

.continue_test_row:
        djnz    .test_char

        ; Add line endings to the emulator log only; the screen has already
        ; advanced by the console's normal 42-column automatic wrap.
        ld      a, 13
        out     (#EF), a
        ld      a, 10
        out     (#EF), a

        ld      a, (bios_dbg_test_row)
        inc     a
        ld      (bios_dbg_test_row), a
        cp      BIOS_DEBUG_ROWS
        jr      nc, .test_done
        ld      a, (bios_dbg_test_char)
        inc     a
        ld      (bios_dbg_test_char), a
        jr      .test_row
.test_done:
        xor     a
        ld      (bios_dbg_no_wrap), a
        ret

; IN: HL=16-bit value. Render four hexadecimal digits in the 6x8 console.
BIOS_DebugConsole_PrintHex16::
        push    af
        push    bc
        ld      a, h
        call    bios_dbg_print_hex8
        ld      a, l
        call    bios_dbg_print_hex8
        pop     bc
        pop     af
        ret

bios_dbg_print_hex8:
        push    af
        rra
        rra
        rra
        rra
        call    bios_dbg_print_nibble
        pop     af
bios_dbg_print_nibble:
        and     #0F
        cp      10
        jr      c, .hex_digit
        add     a, 7
.hex_digit:
        add     a, '0'
        jp      BIOS_DebugConsole_PrintChar

; Advance one 8-pixel character row and scroll the existing lower window.
bios_dbg_newline:
        ld      a, (bios_dbg_row)
        inc     a
        cp      BIOS_DEBUG_ROWS
        jr      c, .row_in_range
        call    UI_ScrollConsoleUp
        ld      a, BIOS_DEBUG_ROWS - 1
.row_in_range:
        ld      (bios_dbg_row), a
        ret

; IN: A=absolute Y (0..191), C=screen byte X (0..31).
; OUT: HL=Spectrum bitmap address for that scanline and byte.
bios_dbg_calc_addr:
        ld      d, a
        and     #C0
        rrca
        rrca
        rrca
        ld      h, a
        ld      a, d
        and     #07
        or      h
        or      #40
        ld      h, a
        ld      a, d
        and     #38
        rlca
        rlca
        add     a, c
        ld      l, a
        ; Start each pixel with a fresh mask = 80h >> (X & 7).
        ld      a, #80
        ld      (bios_dbg_mask), a
        ret

; --- ТAБЛИЧНЫЙ СKРOЛЛИНГ НИЖНЕГO OKНA ЛOГOВ ИЗ ПРOЕKТA 123 ---
UI_ScrollConsoleUp::
        PUSH    IX: PUSH AF: PUSH BC: PUSH DE: PUSH HL
        XOR     A                       
        LD      (.local_micro_ctr), A
.global_loop:
        LD      IX, .Line_Addr_Table
        LD      A, 11                   
        LD      (.local_row_ctr), A       
.rows_loop:
        LD      E, (IX+0): LD D, (IX+1)               
        LD      L, (IX+2): LD H, (IX+3)               
        PUSH    IX: PUSH BC                      
        LD      A, (.local_micro_ctr): ADD A, D: LD D, A                    
        LD      A, (.local_micro_ctr): ADD A, H: LD H, A                    
        LD      BC, 32: LDIR                            
        POP     BC: POP IX                      
        LD      DE, 4: ADD IX, DE                  
        LD      A, (.local_row_ctr): DEC A: LD (.local_row_ctr), A
        JR      NZ, .rows_loop   

        LD      A, (.local_micro_ctr): INC A: LD (.local_micro_ctr), A
        CP      8                        
        JR      C, .global_loop   

        ; Сдвиг цветовых атрибутов лога
        LD      DE, #5980: LD HL, #59A0: LD BC, 352: LDIR                            
        ; Очистка последней строки атрибутов
        LD      HL, #5AE0: LD DE, #5AE1: LD (HL), %01000100: LD BC, 31: LDIR                            
        ; Очистка пикселей 23-й строки
        XOR     A: LD H, #50                  
.clear_pixels:
        PUSH    AF: PUSH HL                      
        LD      L, #E0: LD C, 32: XOR A
.clear_bytes:
        LD      (HL), A: INC HL: DEC C: JR NZ, .clear_bytes  
        POP     HL: INC H: POP AF: INC A: CP 8: JR C, .clear_pixels

        POP     HL: POP DE: POP BC: POP AF: POP IX
        LD      A, 23: LD (Console_Y), A
        XOR     A: LD (Console_X), A
        RET

.Line_Addr_Table:
        DW #4880, #48A0, #48A0, #48C0, #48C0, #48E0, #48E0, #5000                 
        DW #5000, #5020, #5020, #5040, #5040, #5060, #5060, #5080                 
        DW #5080, #50A0, #50A0, #50C0, #50C0, #50E0                 

.local_micro_ctr: DB 0
.local_row_ctr:   DB 0

ClearScreenROM::
        PUSH    BC: PUSH DE: PUSH HL
        LD      HL, #4000: LD DE, #4001: LD (HL), 0: LD BC, 6143: LDIR                            
        LD      HL, #5800: LD DE, #5801: LD (HL), %01000100: LD BC, 767: LDIR
        POP     HL: POP DE: POP BC
        RET

ForceInterfaceColors::
        PUSH    AF: PUSH BC: PUSH DE: PUSH HL
        LD      HL, #5800
        LD      DE, #5801
        LD      (HL), %01001111         ; Верх: Синий
        LD      BC, 383                 
        LDIR
        
        LD      HL, #5980
        LD      DE, #5981
        LD      (HL), %01000100         ; Низ: Черный
        LD      BC, 383                 
        LDIR
        POP     HL: POP DE: POP BC: POP AF
        RET

; --- ТАБЛИЦА АДРЕСОВ АТРИБУТОВ СТРОК КУРСOРА ПАНЕЛИ (Y = 2..10) ---
Cursor_Row_Table:
        DW #5841 , #5861 , #5881 , #58A1 , #58C1 , #58E1 , #5901 , #5921 , #5941

; --- ВЫЧИСЛЕНИЕ АДРЕСА СТРОКИ АТРИБУТОВ ДЛЯ КУРСОРНОЙ НАВИГАЦИИ ---
Get_Cursor_RAM_Address:
        LD      A, (CursorY)            
        SUB     2                       
        ADD     A, A                    
        LD      E, A 
        LD      D, 0 
        LD      HL, Cursor_Row_Table    
        ADD     HL, DE                  
        LD      A, (HL) 
        LD      E, A 
        INC     HL 
        LD      A, (HL) 
        LD      D, A 
        EX      DE, HL                  
        RET 

; --- ОТРЕСОВКА ИНВЕРСИОННОЙ ПОЛОСЫ ВЫДЕЛЕНИЯ ФАЙЛА ---
DrawActiveCursor::
        PUSH    BC                      
        PUSH    HL                      
        CALL    Get_Cursor_RAM_Address  
        LD      B, 30                   
.loop:
        LD      (HL), %00111000         ; Инверсия: Белый фон, черный text
        INC     HL                     
        DJNZ    .loop                  
        POP     HL                     
        POP     BC                      
        RET                             

; --- СТИРАНИЕ ИНВЕРСИОННОЙ ПОЛОСЫ С ВОЗВРАТОМ СИНЕГО ЦВЕТА ---
EraseActiveCursor::
        PUSH    BC                      
        PUSH    HL                      
        CALL    Get_Cursor_RAM_Address  
        LD      B, 30                   
.loop:
        LD      (HL), %01001111         ; Синий фон %4F
        INC     HL                     
        DJNZ    .loop                  
        POP     HL                     
        POP     BC                      
        RET                             

; --- ГЛОБАЛЬНАЯ ТОЧКА ИНИЦИАЛИЗАЦИИ ИНТЕРФЕЙСА ОКОН PERFECT COMMANDER ---
INIT_PROGRAM_INTERFACE::
        PUSH    AF: PUSH BC: PUSH DE: PUSH HL

        CALL    ClearScreenROM          ; Полная очистка экрана и атрибутов
        
        LD      A, 12                   
        LD      (#5C6A), A              ; DF_SZ = 12 строк для панели файлов
        LD      A, %00111000            
        LD      (#5C8D), A             
        LD      A, %01000100            
        LD      (#5C8C), A             

        XOR     A                       
        OUT     (254), A                ; Черный бордюр

        LD      A, %00111000            
        LD      (#5C48), A             
        LD      (#5C8F), A             
        LD      (#5C49), A             

        CALL    ForceInterfaceColors    ; Заливка окон цветом
        CALL    DrawCommanderPanels     ; Отрисовка разделительной линии панелей

        ; Начальные координаты для вывода логов в нижнем черном окне
        LD      A, 12                   
        LD      (Console_Y), A
        XOR     A
        LD      (Console_X), A         

        ; ОТРЕНДЕРИТЬ ДИНАМИЧЕСКИЙ КАТАЛОГ ДИСКА А
        CALL    Refresh_Panel_Directory 

        ; Фиксируем полосу курсора на первую строку файлов (строка 2)
        LD      A, 2                  
        LD      (CursorY), A          
        CALL    DrawActiveCursor       

        POP     HL: POP DE: POP BC: POP AF
        RET

EnsureConsoleSpace::
        PUSH    AF
        PUSH    BC
        PUSH    DE
        PUSH    HL

        LD      A, (Console_Y)
        CP      24                      ; Нижний лог занимает строки 12..23
        JR      C, .space_ok            

        CALL    UI_ScrollConsoleUp      ; Вызываем табличный аппаратный сдвиг экрана
        
        LD      A, 23                   
        LD      (Console_Y), A
        XOR     A
        LD      (Console_X), A          

.space_ok:
        POP     HL
        POP     DE
        POP     BC
        POP     AF
        RET

; PROJECT 124 — TR-DOS directory utilities

; IN: CursorY = visible row (2..10)
; OUT: HL = 16-byte descriptor in TRDOS_PANEL_BUFFER, carry clear.
;      HL=0 and carry set for an invalid/empty row.
GetSelectedFileDescriptor::
        ld      a, (CursorY)
        cp      2
        jr      c, .bad_row
        cp      11
        jr      nc, .bad_row
        sub     2
        ld      l, a
        ld      h, 0
        add     hl, hl
        add     hl, hl
        add     hl, hl
        add     hl, hl
        ld      de, TRDOS_PANEL_BUFFER
        add     hl, de

        ; Reject an empty/deleted/invalid catalogue slot.
        ld      a, (hl)
        cp      32
        jr      c, .bad_row
        cp      127
        jr      nc, .bad_row
        or      a
        ret
.bad_row:
        ld      hl, 0
        scf
        ret

LogSelectedFile::
        push    hl
        ld      hl, .str_exec_prefix
        call    LOG_WriteMessageToBuffer
        pop     hl
        ret
.str_exec_prefix:
        defb "[LAUNCH] Invoking system CODE loader for selected workspace node...", #0D, #0A, 0

; Read and render the complete 128-entry TR-DOS catalogue (8 * 256 bytes).
PrintDiskDirectory::
        push    af
        push    bc
        push    de
        push    hl

        ; Select drive A and clear the two-byte TR-DOS error result.
        xor     a
        ld      (#5D19), a
        ld      (#5D0F), a
        ld      (#5D10), a

        ; C=#05 raw resident convention: D=track 0, E=sector 0..15.
        ld      de, #0000
        ld      b, 8
        ld      c, 5
        ld      hl, TRDOS_PANEL_BUFFER
        call    #3D13

        ; C=#05 reports read errors in the TR-DOS error byte #5D0F.
        ld      a, (#5D0F)
        ld      b, a
        ld      a, (#5D10)
        or      b
        jp      nz, .directory_read_error

        ld      b, 128
        ld      hl, TRDOS_PANEL_BUFFER
        ld      a, 2
        ld      (File_Y), a

.loop_slots:
        push    bc
        push    hl

        ld      a, (hl)
        or      a
        jp      z, .empty_slot
        cp      1
        jp      z, .invalid_slot
        cp      32
        jp      c, .invalid_slot
        cp      127
        jp      nc, .invalid_slot

        ld      a, 2
        ld      (File_X), a
        ld      c, 25
.clear_pixels_loop:
        push    bc
        push    hl
        ld      a, (File_X)
        ld      c, a
        ld      a, (File_Y)
        ld      d, a
        ld      b, " "
        call    Direct_PrintChar_Args
        pop     hl
        pop     bc
        ld      a, (File_X)
        inc     a
        ld      (File_X), a
        dec     c
        jr      nz, .clear_pixels_loop

        ld      a, 2
        ld      (File_X), a
        ld      c, 8
.print_name:
        ld      a, (hl)
        push    bc
        push    hl
        ld      b, a
        ld      a, (File_X)
        ld      c, a
        ld      a, (File_Y)
        ld      d, a
        call    Direct_PrintChar_Args
        ld      a, (File_X)
        inc     a
        ld      (File_X), a
        pop     hl
        pop     bc
        inc     hl
        dec     c
        jr      nz, .print_name

        ; Extension separator at X=10.
        ld      a, 10
        ld      (File_X), a
        push    hl
        ld      c, 10
        ld      a, (File_Y)
        ld      d, a
        ld      b, "."
        call    Direct_PrintChar_Args
        pop     hl

        ; File type at X=11.
        ld      a, 11
        ld      (File_X), a
        ld      a, (hl)
        push    hl
        ld      b, a
        ld      a, 11
        ld      c, a
        ld      a, (File_Y)
        ld      d, a
        call    Direct_PrintChar_Args
        pop     hl

        ; Descriptor byte 13 = allocated sector count. Display ceil(sectors/4)
        ; without 8-bit overflow: (sectors + 3) is calculated in 16 bits.
        push    hl
        ld      de, 13
        add     hl, de
        ld      a, (hl)
        ld      e, a
        ld      d, 0
        ld      hl, 3
        add     hl, de
        srl     h
        rr      l
        srl     h
        rr      l
        ld      a, l
        pop     hl

        ld      c, 16
        call    Util_PrintDecimalByte

        ld      a, 20
        ld      (File_X), a
        ld      a, (File_Y)
        ld      d, a
        ld      c, 20
        ld      b, "K"
        call    Direct_PrintChar_Args

        ld      a, 21
        ld      (File_X), a
        ld      a, (File_Y)
        ld      d, a
        ld      c, 21
        ld      b, "b"
        call    Direct_PrintChar_Args

        ld      a, (File_Y)
        inc     a
        ld      (File_Y), a
        cp      11
        jp      nc, .directory_done

        ; Advance to next 16-byte descriptor. Restore saved slot counter.
        pop     hl
        ld      de, 16
        add     hl, de
        pop     bc
        dec     b
        jp      nz, .loop_slots
        jp      .directory_success

.invalid_slot:
        ; Drop the per-slot saves, advance HL, and decrement the ORIGINAL
        ; 128-entry counter. Do not load 16 into BC (that destroys B), and
        ; do not push the per-slot saves again before re-entering the loop.
        pop     hl
        pop     bc
        ld      de, 16
        add     hl, de
        dec     b
        jp      nz, .loop_slots
        jp      .directory_success

.empty_slot:
        ; Per-slot BC/HL saves are still present.
        pop     hl
        pop     bc
        jp      .directory_success

.directory_done:
        ; Per-slot BC/HL saves are still present.
        pop     hl
        pop     bc
.directory_success:
        pop     hl
        pop     de
        pop     bc
        pop     af
        or      a
        ret

.directory_read_error:
        pop     hl
        pop     de
        pop     bc
        pop     af
        scf
        ret

; IN: A=value 0..255, C=screen X. Prints exactly three decimal digits.
Util_PrintDecimalByte:
        push    af
        push    bc
        push    de
        push    hl
        ld      e, a
        ld      a, c
        ld      (File_X), a
        ld      a, e
        ld      b, 0
.hundreds:
        cp      100
        jr      c, .p_hundreds
        sub     100
        inc     b
        jr      .hundreds
.p_hundreds:
        ld      e, a
        ld      a, b
        add     a, #30
        push    de
        ld      b, a
        ld      a, (File_X)
        ld      c, a
        ld      a, (File_Y)
        ld      d, a
        call    Direct_PrintChar_Args
        ld      a, (File_X)
        inc     a
        ld      (File_X), a
        pop     de

        ld      a, e
        ld      b, 0
.tens:
        cp      10
        jr      c, .p_tens
        sub     10
        inc     b
        jr      .tens
.p_tens:
        ld      e, a
        ld      a, b
        add     a, #30
        push    de
        ld      b, a
        ld      a, (File_X)
        ld      c, a
        ld      a, (File_Y)
        ld      d, a
        call    Direct_PrintChar_Args
        ld      a, (File_X)
        inc     a
        ld      (File_X), a
        pop     de

        ld      a, e
        add     a, #30
        ld      b, a
        ld      a, (File_X)
        ld      c, a
        ld      a, (File_Y)
        ld      d, a
        call    Direct_PrintChar_Args
        pop     hl
        pop     de
        pop     bc
        pop     af
        ret

Refresh_Panel_Directory::
        push    af
        push    bc
        push    de
        push    hl

        ; Fill 320 attribute bytes with #4F. The original byte sequence was
        ; LD HL,#5841, leaving DE undefined. This is the correct fill idiom.
        ld      hl, #5840
        ld      (hl), %01001111
        ld      de, #5841
        ld      bc, 319
        ldir

        call    PrintDiskDirectory
        jr      c, .refresh_failed
        call    DrawActiveCursor
        pop     hl
        pop     de
        pop     bc
        pop     af
        or      a
        ret
.refresh_failed:
        pop     hl
        pop     de
        pop     bc
        pop     af
        scf
        ret

; MEMORY MAP / BOOT:
; The stock GS105a ROM loads a tiny module at $5B00. That module enables

; therefore logical $5000 maps to physical linear address $00D000.
; The x86 DMA frame is placed at $5000: [length lo][length hi][x86 bytes].
; The interpreter starts at $5002 and is bounded by the exact file length.

        ORG     $0000

ROM_RETURN_TRAMPOLINE_ADDR EQU $4000
X86_DMA_HEADER_ADDR        EQU $5000
X86_RAM_RUN_ADDR           EQU $5002
DMA_WAIT_LIMIT   EQU #FFFF
DMA_WAIT_OUTER   EQU 8
IPC_WAIT_LIMIT   EQU #FFFF

; Firmware entry vector
        jp      start_neoprogram
        BLOCK   53, 0
        ei
        ret
        BLOCK   198, 0

start_neoprogram:
        di
        im      1
        ld      sp, $3FFF

        ; Set Z80 to the documented maximum guaranteed clock: 20 MHz.
        ; Do not select 24 MHz: the guide explicitly says it is not guaranteed.
        in      a, (GSCFG0)
        and     #CF
        or      C_20MHZ
        out     (GSCFG0), a

        xor     a
        ld      (neo_dma_error), a
        ld      (saved_z80_ax), a
        ld      (saved_z80_ax+1), a
        ld      (saved_x86_ax), a
        ld      (saved_x86_ax+1), a
        ld      (x86_exit_status), a
        ld      (neo_command), a
        ld      (neo_tx_byte), a

        ; Copy the ROM-return trampoline into unoverlaid RAM at $4000.
        ld      hl, rom_return_trampoline_code
        ld      de, ROM_RETURN_TRAMPOLINE_ADDR
        ld      bc, rom_return_trampoline_end - rom_return_trampoline_code
        ldir

        ; Notify the Spectrum only after the new firmware is executing.
        ld      a, NGS_FIRMWARE_READY_ACK
        call    neo_send_data_byte

main_loop:
        ; A command exists only while CBIT (ZXSTAT bit 0) is set.
        in      a, (ZXSTAT)
        bit     B_CBIT, a
        jr      z, main_loop

        ; Read the command byte from ZXCMD. Reading it does NOT clear CBIT.
        in      a, (ZXCMD)
        ld      (neo_command), a

        ; Standard GS #F4 cold restart also returns this custom firmware to ROM.
        cp      GS_CMD_COLD_RESTART
        jp      z, command_return_to_gsrom
        cp      CMD_RESET
        jp      z, command_reset
        cp      CMD_LOAD_X86
        jp      z, command_load_x86
        cp      CMD_RUN_X86
        jp      z, command_run_x86
        cp      CMD_GET_STATUS
        jp      z, command_get_status
        cp      CMD_MEM_TEST
        jp      z, command_mem_test
        cp      CMD_BIOS_POST
        jp      z, command_bios_post
        cp      CMD_INT_TEST
        jp      z, command_int_test
        cp      CMD_INT_GUEST_TEST
        jp      z, command_int_guest_test
        cp      CMD_BIOS_VIDEO_TEST
        jp      z, command_bios_video_test
        cp      CMD_RUN_X86_REAL
        jp      z, command_run_x86_real

        ; Unknown command: release CBIT so the host cannot deadlock.
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

; A #F4 command is also the standard GS105a cold-restart command.
; If our firmware is running, reveal the stock ROM again and jump to its
; reset vector. Leave CBIT asserted: the ROM's cold-start path clears it when
; the standard firmware is ready, so the host's long wait is a real handshake.
command_return_to_gsrom:
        jp      ROM_RETURN_TRAMPOLINE_ADDR

command_reset:
        xor     a
        ld      (neo_dma_error), a
        ld      (x86_exit_status), a
        ld      (saved_z80_ax), a
        ld      (saved_z80_ax+1), a
        ld      (saved_x86_ax), a
        ld      (saved_x86_ax+1), a
        out     (CLRCBIT), a
        jp      main_loop

; -------------------------------------------------------------------------
; CMD_LOAD_X86
; 1) Select ZX DMA module and set physical target $00D000.
; 2) Enable DMA by setting DMA_CST bit 7.
; 3) Clear CBIT: this is the READY indication to the host.
; 4) Wait for host's end marker (#FF written to GSCOM, setting CBIT).
; 5) Disable DMA, then clear CBIT to acknowledge completion.
;
; Host uses OUT/LDIR writes into $0000-$3FFF. No dummy write is performed.
; -------------------------------------------------------------------------
command_load_x86:
        ld      a, C_DMA_ZX
        out     (DMA_MOD), a

        xor     a
        out     (DMA_HAD), a            ; physical address bits 21:16 = 0
        ld      a, #D0
        out     (DMA_MAD), a            ; physical address bits 15:8 = #D0
        xor     a
        out     (DMA_LAD), a            ; physical address bits 7:0 = 0

        ld      a, #80
        out     (DMA_CST), a            ; DMA enable = bit 7

        xor     a
        out     (CLRCBIT), a            ; host may now start LDIR

        ld      d, DMA_WAIT_OUTER
.dma_wait_outer:
        ld      bc, DMA_WAIT_LIMIT
.dma_wait_host_end:
        in      a, (ZXSTAT)
        bit     B_CBIT, a
        jr      nz, .dma_host_end
        dec     bc
        ld      a, b
        or      c
        jr      nz, .dma_wait_host_end
        dec     d
        jr      nz, .dma_wait_outer

        ; Timeout: close DMA and release the host command line.
        xor     a
        out     (DMA_CST), a
        out     (CLRCBIT), a
        ld      a, 1
        ld      (neo_dma_error), a
        ld      a, 3
        ld      (x86_exit_status), a
        jp      main_loop

.dma_host_end:
        xor     a
        out     (DMA_CST), a            ; disable DMA before leaving handler
        out     (CLRCBIT), a            ; acknowledge host's #FF marker
        xor     a
        ld      (neo_dma_error), a
        jp      main_loop

; -------------------------------------------------------------------------
; CMD_RUN_X86 — execute the length-framed payload beginning at $5002.
; -------------------------------------------------------------------------
command_run_x86:
        call    execute_neogs_x86
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

execute_neogs_x86:
        ld      a, (neo_dma_error)
        or      a
        jr      z, .dma_ok
        ld      hl, $FFFF
        ld      (saved_x86_ax), hl
        ld      a, 3
        ld      (x86_exit_status), a
        ret

.dma_ok:
        ; DMA frame begins with a little-endian x86 byte length at $5000.
        ld      hl, (X86_DMA_HEADER_ADDR)
        ld      a, h
        or      l
        jp      z, .invalid_x86_length

        ; Maximum code length is $2FFE: $5002 + length must not exceed $8000.
        ld      a, h
        cp      #2F
        jr      c, .x86_length_valid
        jp      nz, .invalid_x86_length
        ld      a, l
        cp      #FF
        jp      nc, .invalid_x86_length

.x86_length_valid:
        ld      de, X86_RAM_RUN_ADDR
        add     hl, de
        ld      (x86_program_end), hl

        ld      hl, 0
        ld      (saved_z80_ax), hl
        ld      hl, #FFFF
        ld      (x86_steps_remaining), hl
        xor     a
        ld      (x86_exit_status), a

        ld      bc, X86_RAM_RUN_ADDR
        call    x86_execute_loop

        ld      hl, (saved_z80_ax)
        ld      (saved_x86_ax), hl
        ret

.invalid_x86_length:
        ld      hl, #FFFF
        ld      (saved_z80_ax), hl
        ld      (saved_x86_ax), hl
        ld      a, 4
        ld      (x86_exit_status), a
        ret

; Execute the loaded DMA task as bounded guest code at 0000:5002.
; The DMA frame is copied into the x86 physical-memory backend before execution.
execute_neogs_x86_real:
        ; Match the legacy loader's DMA error and length validation.
        ld      a, (neo_dma_error)
        or      a
        jr      z, .real_dma_ok
        ld      hl, #FFFF
        ld      (saved_x86_ax), hl
        ld      a, 3
        ld      (x86_exit_status), a
        scf
        ret

.real_dma_ok:
        ld      hl, (X86_DMA_HEADER_ADDR)
        ld      a, h
        or      l
        jp      z, .real_invalid_length
        ld      a, h
        cp      #2F
        jr      c, .real_length_valid
        jp      nz, .real_invalid_length
        ld      a, l
        cp      #FF
        jp      nc, .real_invalid_length

.real_length_valid:
        ld      de, X86_RAM_RUN_ADDR
        add     hl, de
        ld      (x86_real_task_end), hl

        call    x86_mem_init
        ld      hl, X86_RAM_RUN_ADDR
        ld      de, #5002
        ld      bc, (X86_DMA_HEADER_ADDR)
        call    x86_real_copy_image
        jp      c, .real_copy_failed

        ; Reset the guest register file to a deterministic real-mode entry.
        ld      hl, 0
        ld      (x86_reg_ax), hl
        ld      (x86_reg_bx), hl
        ld      (x86_reg_cx), hl
        ld      (x86_reg_dx), hl
        ld      (x86_reg_bp), hl
        ld      (x86_reg_si), hl
        ld      (x86_reg_di), hl
        ld      (x86_reg_es), hl
        ld      (x86_reg_ds), hl
        ld      (x86_reg_cs), hl
        ld      hl, #5002
        ld      (x86_reg_ip), hl
        ld      hl, #7000
        ld      (x86_reg_ss), hl
        ld      hl, #0100
        ld      (x86_reg_sp), hl
        ld      hl, #0202
        ld      (x86_reg_flags), hl

        ld      a, 1
        ld      (x86_real_bounds_enable), a
        call    x86_real_execute
        push    af
        ld      hl, (x86_reg_ax)
        ld      (saved_x86_ax), hl
        xor     a
        ld      (x86_real_bounds_enable), a
        pop     af
        ret

.real_invalid_length:
        ld      hl, #FFFF
        ld      (saved_x86_ax), hl
        ld      a, 4
        ld      (x86_exit_status), a
        scf
        ret

.real_copy_failed:
        ld      hl, #1D01
        ld      (saved_x86_ax), hl
        ld      a, 4
        ld      (x86_exit_status), a
        scf
        ret

; -------------------------------------------------------------------------
; CMD_MEM_TEST — destructive test of the reserved 1 MiB x86 RAM region.
; Result is returned through the existing CMD_GET_STATUS frame.
; -------------------------------------------------------------------------
command_mem_test:
        call    x86_mem_test
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

; BIOS low-memory foundation POST.
command_bios_post:
        call    bios_post
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

; Stage 2A: validate 8086 INT/IRET stack-frame semantics.
command_int_test:
        call    x86_int_self_test
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

; Stage 2B: execute a guest INT handler through CS:IP and return with IRET.
command_int_guest_test:
        call    x86_real_mode_test
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

; Stage 3A: guest INT 10h video BIOS service test.
command_bios_video_test:
        call    bios_int10_video_test
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

; Run a TR-DOS payload through the bounded CS:IP real-mode executor.
command_run_x86_real:
        call    execute_neogs_x86_real
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

; -------------------------------------------------------------------------
; CMD_GET_STATUS — application-level frame over the documented GSDAT stream:
;   A5 signature, exit status, AX low, AX high.
; Every byte uses DBIT: write ZXDATWR -> host reads GSDAT -> DBIT clears.
; -------------------------------------------------------------------------
command_get_status:
        ld      a, NGS_RESULT_SIGNATURE
        call    neo_send_data_byte
        jr      c, .status_tx_failed

        ld      a, (x86_exit_status)
        call    neo_send_data_byte
        jr      c, .status_tx_failed

        ld      hl, (saved_x86_ax)
        ld      a, l
        call    neo_send_data_byte
        jr      c, .status_tx_failed

        ld      a, h
        call    neo_send_data_byte
        jr      c, .status_tx_failed

        xor     a
        out     (CLRCBIT), a
        jp      main_loop

.status_tx_failed:
        xor     a
        out     (CLRCBIT), a
        jp      main_loop

; -------------------------------------------------------------------------
; neo_send_data_byte
; IN: A = byte to send to Spectrum.
; Wait for DBIT=0, write ZXDATWR (#03), then wait for host read to clear DBIT.
; OUT: carry clear on success, carry set on timeout. Preserves HL and DE.
; -------------------------------------------------------------------------
neo_send_data_byte:
        ld      (neo_tx_byte), a
        ld      bc, IPC_WAIT_LIMIT
.tx_wait_slot:
        in      a, (ZXSTAT)
        bit     B_DBIT, a
        jr      z, .tx_slot_free
        dec     bc
        ld      a, b
        or      c
        jr      nz, .tx_wait_slot
        scf
        ret

.tx_slot_free:
        ld      a, (neo_tx_byte)
        out     (ZXDATWR), a

        ld      bc, IPC_WAIT_LIMIT
.tx_wait_host_read:
        in      a, (ZXSTAT)
        bit     B_DBIT, a
        jr      z, .tx_host_read
        dec     bc
        ld      a, b
        or      c
        jr      nz, .tx_wait_host_read
        scf
        ret

.tx_host_read:
        or      a
        ret

neo_command:       db 0
neo_tx_byte:       db 0
neo_dma_error:     db 0
saved_z80_ax:      dw 0
saved_x86_ax:      dw 0

; This code is copied to $4000 during startup. Its OUT changes the ROM map,
; but the following JP is still fetched from RAM because PC remains in $4000.
rom_return_trampoline_code:
        in      a, (GSCFG0)
        and     #FE
        out     (GSCFG0), a
        jp      0
rom_return_trampoline_end:

        INCLUDE "x86_memory.asm"
        INCLUDE "bios/bios_core.asm"
        INCLUDE "x86_modrm.asm"
        INCLUDE "x86_interpreter.asm"
        INCLUDE "x86_interrupt.asm"
        INCLUDE "x86_real_mode.asm"
        INCLUDE "bios/bios_video.asm"

        ASSERT ROM_RETURN_TRAMPOLINE_ADDR + 16 <= X86_DMA_HEADER_ADDR
        ASSERT X86_RAM_RUN_ADDR + NGS_X86_MAX_CODE_SIZE = #8000

neogs_firmware_end:


; =========================================================================
; FILE: neogs_sub/bootstrap.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — first-stage NeoGS executable bootstrap
;
; GS105a command #14 (RECEIVE BLOCK) loads this raw Z80 image into fixed RAM
; at $4000. GS105a command #13 (JUMP TO ADDRESS) enters it at $4000.
;
; The bootstrap maps RAM over the low ROM window, receives a 16-bit firmware
; length and exactly that many bytes through the documented ZX data channel,
; stores the custom firmware at RAM address $0000, then jumps to $0000.
;
; It executes from the fixed RAM window $4000-$7FFF and keeps its stack below
; $4000. The fixed $4000-$7FFF RAM window remains visible after B_NOROM is set.
; =========================================================================

neogs_bootstrap_start:
        di
        ld      sp, #3FFF

        ; Hide the stock ROM and expose RAM at logical $0000-$3FFF.
        in      a, (GSCFG0)
        and     #FD                    ; clear RAMRO: low RAM must be writable
        or      M_NOROM                ; map RAM over the stock ROM at $0000
        out     (GSCFG0), a

        ; Tell Spectrum that the tiny loader is running and waiting for length.
        ld      a, NGS_BOOTSTRAP_READY_ACK
        call    bootstrap_send_data_byte

        ; Host sends length low byte, then high byte.
        call    bootstrap_receive_data_byte
        ld      e, a
        call    bootstrap_receive_data_byte
        ld      d, a

        ; Reject zero length and any image larger than the reserved low RAM area.
        ld      a, d
        or      e
        jr      z, bootstrap_error
        ld      a, d
        cp      #30
        jr      c, bootstrap_length_ok
        jr      nz, bootstrap_error
        ld      a, e
        or      a
        jr      nz, bootstrap_error

bootstrap_length_ok:
        ; Explicitly acknowledge that the length is accepted before the host
        ; starts streaming the firmware payload.
        ld      a, NGS_BOOTSTRAP_LENGTH_ACK
        call    bootstrap_send_data_byte
        ld      hl, 0

bootstrap_copy_loop:
        ld      a, d
        or      e
        jr      z, bootstrap_start_firmware
        call    bootstrap_receive_data_byte
        ld      (hl), a
        inc     hl
        dec     de
        jr      bootstrap_copy_loop

bootstrap_start_firmware:
        jp      0

bootstrap_error:
        ld      a, NGS_BOOTSTRAP_ERROR_ACK
        call    bootstrap_send_data_byte
        jp      bootstrap_halt

; Terminal state used when the host stops responding or cannot consume data.
; Never leave a bootstrap polling loop running forever with no timeout.
bootstrap_halt:
        di
.bootstrap_halt_loop:
        halt
        jr      .bootstrap_halt_loop

; Best-effort error ACK after a receive timeout. This path has its own
; bounded waits and never calls bootstrap_send_data_byte recursively.
bootstrap_timeout:
        ld      a, NGS_BOOTSTRAP_ERROR_ACK
        ld      (bootstrap_tx_byte), a
        ld      bc, #FFFF
.timeout_wait_slot:
        in      a, (ZXSTAT)
        bit     B_DBIT, a
        jr      z, .timeout_slot_ready
        dec     bc
        ld      a, b
        or      c
        jr      nz, .timeout_wait_slot
        jp      bootstrap_halt
.timeout_slot_ready:
        ld      a, (bootstrap_tx_byte)
        out     (ZXDATWR), a
        ld      bc, #FFFF
.timeout_wait_host:
        in      a, (ZXSTAT)
        bit     B_DBIT, a
        jr      z, bootstrap_halt
        dec     bc
        ld      a, b
        or      c
        jr      nz, .timeout_wait_host
        jp      bootstrap_halt

; Wait for one byte from Spectrum. Reading ZXDATRD clears DBIT.
bootstrap_receive_data_byte:
        ld      bc, #FFFF
.rx_wait:
        in      a, (ZXSTAT)
        bit     B_DBIT, a
        jr      nz, .rx_ready
        dec     bc
        ld      a, b
        or      c
        jr      nz, .rx_wait
        jp      bootstrap_timeout
.rx_ready:
        in      a, (ZXDATRD)
        ret

; Send one byte to Spectrum. Host reads GSDAT and clears DBIT.
bootstrap_send_data_byte:
        ld      (bootstrap_tx_byte), a
        ld      bc, #FFFF
.tx_free:
        in      a, (ZXSTAT)
        bit     B_DBIT, a
        jr      z, .tx_slot_free
        dec     bc
        ld      a, b
        or      c
        jr      nz, .tx_free
        jp      bootstrap_halt
.tx_slot_free:
        ld      a, (bootstrap_tx_byte)
        out     (ZXDATWR), a
        ld      bc, #FFFF
.tx_wait_read:
        in      a, (ZXSTAT)
        bit     B_DBIT, a
        jr      z, .tx_done
        dec     bc
        ld      a, b
        or      c
        jr      nz, .tx_wait_read
        jp      bootstrap_halt
.tx_done:
        ret

bootstrap_tx_byte:
        db      0

neogs_bootstrap_end:


; =========================================================================
; FILE: neogs_sub/x86_interpreter.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — minimal x86 interpreter core for NeoGS Z80
;
; Supported opcodes:
;   90       NOP
;   40       INC AX (8086 legacy one-byte register encoding)
;   B8 iw    MOV AX, imm16
;   F4       HLT (successful task termination)
;
; Unsupported opcode: status 2. Instruction budget exhausted: status 1.
; DMA error: status 3. Out-of-range/truncated instruction: status 4.
; x86_program_end is the exclusive logical address after the payload.
;
; Entry: BC = x86 IP in NeoGS Z80 logical address space; AX stored in
;        saved_z80_ax. Exit: saved_z80_ax and x86_exit_status are updated.
; =========================================================================

x86_execute_loop:
x86_next_instruction:
        ; Every opcode fetch must remain strictly below the exclusive end.
        ld      hl, (x86_program_end)
        or      a
        sbc     hl, bc
        jp      c, x86_ip_out_of_range
        ld      a, h
        or      l
        jp      z, x86_ip_out_of_range

        ld      hl, (x86_steps_remaining)
        ld      a, h
        or      l
        jr      nz, .budget_ok
        ld      a, 1
        ld      (x86_exit_status), a
        ld      hl, (saved_z80_ax)
        ret

.budget_ok:
        dec     hl
        ld      (x86_steps_remaining), hl

        ; Fetch opcode and advance IP.
        ld      a, (bc)
        inc     bc

        ; Resolve handler from 256-entry WORD table.
        ld      l, a
        ld      h, 0
        add     hl, hl
        ld      de, x86_jump_table
        add     hl, de
        ld      e, (hl)
        inc     hl
        ld      d, (hl)

        ; Restore AX in HL and jump to the selected handler. PUSH/RET is
        ; the Z80's safe indirect-jump idiom; BC remains the virtual x86 IP.
        ld      hl, (saved_z80_ax)
        push    de
        ret

; 90 — NOP
x86_nop:
        ld      (saved_z80_ax), hl
        jp      x86_next_instruction

; 40 — INC AX
x86_inc_ax:
        inc     hl
        ld      (saved_z80_ax), hl
        jp      x86_next_instruction

; B8 iw — MOV AX, imm16 (little-endian immediate)
x86_mov_ax_imm16:
        ; B8 requires two immediate bytes. Reject a truncated instruction.
        ld      hl, (x86_program_end)
        or      a
        sbc     hl, bc
        ld      de, 2
        or      a
        sbc     hl, de
        jp      c, x86_ip_out_of_range

        ld      l, c
        ld      h, b
        ld      e, (hl)
        inc     hl
        ld      d, (hl)
        inc     hl
        ld      b, h
        ld      c, l
        ld      (saved_z80_ax), de
        jp      x86_next_instruction

; F4 — HLT: successful end of task
x86_hlt:
        ld      (saved_z80_ax), hl
        xor     a
        ld      (x86_exit_status), a
        ret

; Unsupported instruction: stop rather than executing arbitrary memory.
x86_unsupported:
        ld      (saved_z80_ax), hl
        ld      a, 2
        ld      (x86_exit_status), a
        ret

x86_ip_out_of_range:
        ld      a, 4
        ld      (x86_exit_status), a
        ld      hl, (saved_z80_ax)
        ret

        ALIGN   256
x86_jump_table:
        DW      x86_unsupported             ; 00
        DW      x86_unsupported             ; 01
        REPT    #40 - #02
        DW      x86_unsupported
        ENDR
        DW      x86_inc_ax                  ; 40
        REPT    #90 - #41
        DW      x86_unsupported
        ENDR
        DW      x86_nop                     ; 90
        REPT    #B8 - #91
        DW      x86_unsupported
        ENDR
        DW      x86_mov_ax_imm16            ; B8
        REPT    #F4 - #B9
        DW      x86_unsupported
        ENDR
        DW      x86_hlt                     ; F4
        REPT    256 - #F5
        DW      x86_unsupported
        ENDR

x86_steps_remaining: dw 0
x86_program_end:     dw 0
x86_exit_status:     db 0


; =========================================================================
; FILE: neogs_sub/x86_memory.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — 8086 physical memory backend for NeoGS GS105a
;
; x86 physical $00000-$FFFFF maps to NeoGS RAM pages 4..67
; (NeoGS physical base $010000, page size $4000 / 16 KiB).
; Window C000-FFFF is switched through MPAGEX. This keeps firmware in
; window 0 / page 0 and the current DMA frame in window 1 / page 3.
;
; x86_mem_addr is a 20-bit little-endian address:
;   +0 = bits 7:0, +1 = bits 15:8, +2 = bits 19:16 (valid 0..0F).
;
; API:
;   x86_mem_init: enable NOROM + EXPAG, preserving other GSCFG0 bits.
;   x86_mem_read8:  IN x86_mem_addr; OUT A=value, carry=invalid address.
;   x86_mem_write8: IN x86_mem_addr and x86_mem_value; carry=invalid address.
;   x86_mem_test: destructive test of the reserved 1 MiB x86 RAM region.
;                 status=0, AX=0001 on pass; status=5, AX=phase:page on fail.
;
; The test leaves NOROM+EXPAG enabled. Only window 3 is switched by MPAGEX.
; =========================================================================

X86_MEM_NGSPAGE_BASE EQU 4
X86_MEM_PAGE_COUNT   EQU 64

x86_mem_init:
        in      a, (GSCFG0)
        or      M_NOROM | M_EXPAG
        out     (GSCFG0), a
        ret

; Convert x86_mem_addr into a Z80 pointer in C000-FFFF.
; OUT: HL=window pointer, carry clear; carry set if address > FFFFF.
x86_mem_map_addr:
        ld      a, (x86_mem_addr+2)
        cp      #10
        jr      nc, .bad_address

        ; page = (physical >> 14) + 4
        ; page index = (high_nibble << 2) | (middle_byte >> 6)
        add     a, a
        add     a, a
        ld      b, a
        ld      a, (x86_mem_addr+1)
        rlca
        rlca
        and     #03
        or      b
        add     a, X86_MEM_NGSPAGE_BASE

        ; MPAGEX uses ROL8(page,1); send ROR8(page,1).
        rrca
        out     (MPAGEX), a

        ; Window C000-FFFF maps the selected 16 KiB physical page.
        ld      a, (x86_mem_addr+1)
        and     #3F
        or      #C0
        ld      h, a
        ld      a, (x86_mem_addr)
        ld      l, a
        or      a
        ret

.bad_address:
        scf
        ret

; Read one byte. IN x86_mem_addr[0..2]. OUT A=byte, carry on invalid address.
x86_mem_read8:
        call    x86_mem_map_addr
        ret     c
        ld      a, (hl)
        or      a
        ret

; Write one byte. IN x86_mem_addr[0..2], x86_mem_value.
x86_mem_write8:
        call    x86_mem_map_addr
        ret     c
        ld      a, (x86_mem_value)
        ld      (hl), a
        or      a
        ret

; -------------------------------------------------------------------------
; Destructive physical RAM mapping test.
; Every page receives unique values at offsets 0000 and 3FFF. A second pass
; verifies all 128 locations, detecting missing pages and page aliases.
; -------------------------------------------------------------------------
x86_mem_test:
        call    x86_mem_init
        xor     a
        ld      (x86_mem_page), a
        ld      (x86_mem_phase), a

.write_page_loop:
        call    x86_mem_build_page_start
        ld      a, (x86_mem_page)
        xor     #5A
        ld      (x86_mem_value), a
        call    x86_mem_write8
        jp      c, x86_mem_test_write_start_fail

        call    x86_mem_build_page_end
        ld      a, (x86_mem_page)
        xor     #A5
        ld      (x86_mem_value), a
        call    x86_mem_write8
        jp      c, x86_mem_test_write_end_fail

        ld      a, (x86_mem_page)
        inc     a
        ld      (x86_mem_page), a
        cp      X86_MEM_PAGE_COUNT
        jr      c, .write_page_loop

        xor     a
        ld      (x86_mem_page), a

.verify_page_loop:
        call    x86_mem_build_page_start
        call    x86_mem_read8
        jp      c, x86_mem_test_verify_start_fail
        ld      b, a
        ld      a, (x86_mem_page)
        xor     #5A
        cp      b
        jp      nz, x86_mem_test_verify_start_fail

        call    x86_mem_build_page_end
        call    x86_mem_read8
        jp      c, x86_mem_test_verify_end_fail
        ld      b, a
        ld      a, (x86_mem_page)
        xor     #A5
        cp      b
        jp      nz, x86_mem_test_verify_end_fail

        ld      a, (x86_mem_page)
        inc     a
        ld      (x86_mem_page), a
        cp      X86_MEM_PAGE_COUNT
        jr      c, .verify_page_loop

        ; The physical page test passed. Now verify segment:offset arithmetic,
        ; word access across a 16 KiB page, 16-bit offset wrap, and 20-bit wrap.
        call    x86_mem_word_test
        ret     c

        ; Success: AX=0001 means all memory/addressing tests passed.
        ld      hl, 1
        ld      (saved_x86_ax), hl
        xor     a
        ld      (x86_exit_status), a
        ret

; -------------------------------------------------------------------------
; Segment:offset -> physical address.
; Uses x86_mem_segment and x86_mem_offset_work; result in x86_mem_addr.
; Physical address = ((segment << 4) + offset) & FFFFFh.
; -------------------------------------------------------------------------
x86_mem_calc_phys:
        ; Byte 0 = (segment low << 4) + offset low.
        ld      a, (x86_mem_segment)
        rlca
        rlca
        rlca
        rlca
        ld      b, a
        ld      a, (x86_mem_offset_work)
        add     a, b
        ld      (x86_mem_addr), a
        ld      a, 0
        adc     a, 0
        ld      (x86_mem_carry), a

        ; Byte 1 base = (segment low >> 4) | (segment high << 4).
        ld      a, (x86_mem_segment)
        rrca
        rrca
        rrca
        rrca
        and     #0F
        ld      b, a
        ld      a, (x86_mem_segment+1)
        rlca
        rlca
        rlca
        rlca
        or      b

        ; Add offset high to byte-1 base, then add byte-0 carry.
        ; The two carry-outs are mutually exclusive for byte addition.
        ld      b, a
        ld      a, (x86_mem_offset_work+1)
        add     a, b
        ld      d, a
        ld      a, 0
        adc     a, 0
        ld      (x86_mem_carry1), a   ; carry from base + offset high

        ld      a, d
        ld      b, a
        ld      a, (x86_mem_carry)
        add     a, b
        ld      (x86_mem_addr+1), a
        ld      a, 0
        adc     a, 0
        ld      b, a
        ld      a, (x86_mem_carry1)
        or      b
        ld      (x86_mem_carry), a

        ; Byte 2 is the high segment nibble plus byte-1 carry.
        ld      a, (x86_mem_segment+1)
        rrca
        rrca
        rrca
        rrca
        and     #0F
        ld      b, a
        ld      a, (x86_mem_carry)
        add     a, b
        and     #0F                 ; 20-bit physical address wrap
        ld      (x86_mem_addr+2), a
        ret

; Read a little-endian word at segment:offset.
; IN: x86_mem_segment, x86_mem_offset. OUT: HL=word, carry=invalid.
; The second byte uses (offset+1) modulo 65536 in the SAME segment.
x86_mem_read16:
        ld      hl, (x86_mem_offset)
        ld      (x86_mem_offset_work), hl
        call    x86_mem_calc_phys
        call    x86_mem_read8
        ret     c
        ld      e, a

        ld      hl, (x86_mem_offset_work)
        inc     hl
        ld      (x86_mem_offset_work), hl
        call    x86_mem_calc_phys
        call    x86_mem_read8
        ret     c
        ld      h, a
        ld      l, e
        or      a
        ret

; Write a little-endian word at segment:offset.
; IN: x86_mem_segment, x86_mem_offset, x86_mem_word_value.
; The second byte uses (offset+1) modulo 65536 in the SAME segment.
x86_mem_write16:
        ld      hl, (x86_mem_offset)
        ld      (x86_mem_offset_work), hl
        call    x86_mem_calc_phys
        ld      a, (x86_mem_word_value)
        ld      (x86_mem_value), a
        call    x86_mem_write8
        ret     c

        ld      hl, (x86_mem_offset_work)
        inc     hl
        ld      (x86_mem_offset_work), hl
        call    x86_mem_calc_phys
        ld      a, (x86_mem_word_value+1)
        ld      (x86_mem_value), a
        call    x86_mem_write8
        ret

; Addressing self-test. Uses only the reserved x86 RAM pages 4..67.
; Case 0 crosses a 16 KiB page boundary; case 1 wraps 16-bit offset;
; case 2 wraps the 20-bit physical address space.
x86_mem_word_test:
        ld      hl, #1000
        ld      (x86_mem_segment), hl
        ld      hl, #3FFF
        ld      (x86_mem_offset), hl
        ld      hl, #A55A
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        jp      c, x86_mem_word_fail_0
        call    x86_mem_read16
        jp      c, x86_mem_word_fail_0
        ld      de, #A55A
        or      a
        sbc     hl, de
        jp      nz, x86_mem_word_fail_0

        ld      hl, #1000
        ld      (x86_mem_segment), hl
        ld      hl, #FFFF
        ld      (x86_mem_offset), hl
        ld      hl, #5AA5
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        jp      c, x86_mem_word_fail_1
        call    x86_mem_read16
        jp      c, x86_mem_word_fail_1
        ld      de, #5AA5
        or      a
        sbc     hl, de
        jp      nz, x86_mem_word_fail_1

        ld      hl, #FFFF
        ld      (x86_mem_segment), hl
        ld      (x86_mem_offset), hl
        ld      hl, #C33C
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        jp      c, x86_mem_word_fail_2
        call    x86_mem_read16
        jp      c, x86_mem_word_fail_2
        ld      de, #C33C
        or      a
        sbc     hl, de
        jp      nz, x86_mem_word_fail_2

        or      a
        ret

x86_mem_word_fail_0:
        ld      a, 5
        jr      x86_mem_word_fail
x86_mem_word_fail_1:
        ld      a, 6
        jr      x86_mem_word_fail
x86_mem_word_fail_2:
        ld      a, 7
x86_mem_word_fail:
        ld      (x86_mem_phase), a
        ld      a, (x86_mem_phase)
        ld      h, a
        ld      l, 0
        ld      (saved_x86_ax), hl
        ld      a, 6
        ld      (x86_exit_status), a
        scf
        ret

x86_mem_test_write_start_fail:
        ld      a, 1
        jr      x86_mem_test_fail
x86_mem_test_write_end_fail:
        ld      a, 2
        jr      x86_mem_test_fail
x86_mem_test_verify_start_fail:
        ld      a, 3
        jr      x86_mem_test_fail
x86_mem_test_verify_end_fail:
        ld      a, 4

x86_mem_test_fail:
        ld      (x86_mem_phase), a
        ld      a, (x86_mem_page)
        ld      l, a
        ld      a, (x86_mem_phase)
        ld      h, a
        ld      (saved_x86_ax), hl
        ld      a, 5
        ld      (x86_exit_status), a
        ret

; Build page * 4000h, offset 0000.
x86_mem_build_page_start:
        xor     a
        ld      (x86_mem_addr), a
        ld      a, (x86_mem_page)
        ld      b, a
        and     #03
        add     a, a
        add     a, a
        add     a, a
        add     a, a
        add     a, a
        add     a, a
        ld      (x86_mem_addr+1), a
        ld      a, b
        rrca
        rrca
        and     #0F
        ld      (x86_mem_addr+2), a
        ret

; Build page * 4000h + 3FFFh.
x86_mem_build_page_end:
        ld      a, #FF
        ld      (x86_mem_addr), a
        ld      a, (x86_mem_page)
        ld      b, a
        and     #03
        add     a, a
        add     a, a
        add     a, a
        add     a, a
        add     a, a
        add     a, a
        or      #3F
        ld      (x86_mem_addr+1), a
        ld      a, b
        rrca
        rrca
        and     #0F
        ld      (x86_mem_addr+2), a
        ret

x86_mem_addr:  db 0, 0, 0
x86_mem_value: db 0
x86_mem_page:  db 0
x86_mem_phase: db 0
x86_mem_segment: dw 0
x86_mem_offset: dw 0
x86_mem_offset_work: dw 0
x86_mem_word_value: dw 0
x86_mem_carry: db 0
x86_mem_carry1: db 0


; =========================================================================
; FILE: neogs_sub/bios/bios_core.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — BIOS foundation / POST, stage 1
;
; Establishes the guest-visible low-memory data layout for a future 8086
; real-mode BIOS. This stage does not implement executable BIOS ROM or INT
; services. IVT is initialized to zero; vectors are installed only after the
; CPU INT/IRET path and guest stack semantics are implemented.
;
; Physical memory profile:
;   00000-003FF  IVT (256 entries, 4 bytes each)
;   00400-004FF  BIOS Data Area (BDA)
;   00500-9FFFF  conventional RAM profile (640 KiB)
;   A0000-BFFFF  reserved video memory
;   C0000-EFFFF  reserved option-ROM/device window
;   F0000-FFFFF  reserved P124 BIOS ROM (not read-only yet)
;
; Entry: bios_post
; Success: x86_exit_status=0, saved_x86_ax=#B105
; Failure: x86_exit_status=7, saved_x86_ax=#B1xx
; No DMA/IPC transport changes.
; =========================================================================

BIOS_IVT_BASE             EQU #00000
BIOS_IVT_SIZE             EQU #00400
BIOS_BDA_BASE             EQU #00400
BIOS_BDA_SIZE             EQU #00100
BIOS_LOW_MEMORY_END       EQU #00500
BIOS_CONVENTIONAL_KB      EQU 640
BIOS_VIDEO_MODE_TEXT      EQU #03
BIOS_TEXT_COLUMNS         EQU 80
BIOS_TEXT_ROWS            EQU 25
BIOS_POST_SIGNATURE       EQU #B105

bios_post:
        call    x86_mem_init
        call    bios_clear_ivt_bda
        jp      c, bios_post_fail_1

        ; Equipment word: no legacy IBM-PC equipment bits are asserted.
        ld      hl, #0410
        xor     a
        call    bios_write_byte
        jp      c, bios_post_fail_2
        ld      hl, #0411
        xor     a
        call    bios_write_byte
        jp      c, bios_post_fail_2

        ; Conventional memory size = 640 KiB.
        ld      hl, #0413
        ld      a, #80
        call    bios_write_byte
        jp      c, bios_post_fail_3
        ld      hl, #0414
        ld      a, #02
        call    bios_write_byte
        jp      c, bios_post_fail_3

        ; Empty keyboard ring buffer. Head and tail point to offset 001Eh.
        ld      hl, #041A
        ld      a, #1E
        call    bios_write_byte
        jp      c, bios_post_fail_4
        ld      hl, #041B
        xor     a
        call    bios_write_byte
        jp      c, bios_post_fail_4
        ld      hl, #041C
        ld      a, #1E
        call    bios_write_byte
        jp      c, bios_post_fail_4
        ld      hl, #041D
        xor     a
        call    bios_write_byte
        jp      c, bios_post_fail_4

        ; Text mode 03h, 80 columns, 4 KiB page, page zero active.
        ld      hl, #0449
        ld      a, BIOS_VIDEO_MODE_TEXT
        call    bios_write_byte
        jp      c, bios_post_fail_5
        ld      hl, #044A
        ld      a, BIOS_TEXT_COLUMNS
        call    bios_write_byte
        jp      c, bios_post_fail_5
        ld      hl, #044B
        xor     a
        call    bios_write_byte
        jp      c, bios_post_fail_5
        ld      hl, #044C
        xor     a
        call    bios_write_byte
        jp      c, bios_post_fail_5
        ld      hl, #044D
        ld      a, #10
        call    bios_write_byte
        jp      c, bios_post_fail_5
        ld      hl, #0462
        xor     a
        call    bios_write_byte
        jp      c, bios_post_fail_5

        ; 25 rows are represented as rows-minus-one; character height = 16.
        ld      hl, #0484
        ld      a, BIOS_TEXT_ROWS - 1
        call    bios_write_byte
        jp      c, bios_post_fail_6
        ld      hl, #0485
        ld      a, #10
        call    bios_write_byte
        jp      c, bios_post_fail_6
        ld      hl, #0486
        xor     a
        call    bios_write_byte
        jp      c, bios_post_fail_6

        ; Read back every IVT byte, then verify the published BDA profile.
        call    bios_verify_ivt
        jp      c, bios_post_fail_7

        ld      hl, #0410
        xor     a
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #0411
        xor     a
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #0413
        ld      a, #80
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #0414
        ld      a, #02
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #041A
        ld      a, #1E
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #041B
        xor     a
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #041C
        ld      a, #1E
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #041D
        xor     a
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #0449
        ld      a, BIOS_VIDEO_MODE_TEXT
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #044A
        ld      a, BIOS_TEXT_COLUMNS
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #044B
        xor     a
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #044C
        xor     a
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #044D
        ld      a, #10
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #0462
        xor     a
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #0484
        ld      a, BIOS_TEXT_ROWS - 1
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #0485
        ld      a, #10
        call    bios_expect_byte
        jp      c, bios_post_fail_8
        ld      hl, #0486
        xor     a
        call    bios_expect_byte
        jp      c, bios_post_fail_8

        ; Publish the INT 10h firmware trap after the IVT zero-fill check.
        call    bios_int10_install_vector
        jp      c, bios_post_fail_8

        ld      hl, BIOS_POST_SIGNATURE
        ld      (saved_x86_ax), hl
        xor     a
        ld      (x86_exit_status), a
        or      a
        ret

; Clear IVT + BDA (first 1280 bytes).
bios_clear_ivt_bda:
        ld      hl, BIOS_IVT_BASE
        ld      (bios_clear_address), hl
        ld      hl, BIOS_LOW_MEMORY_END
        ld      (bios_clear_remaining), hl
        xor     a
        ld      (x86_mem_value), a

.bios_clear_loop:
        ld      hl, (bios_clear_address)
        ld      (x86_mem_addr), hl
        xor     a
        ld      (x86_mem_addr+2), a
        call    x86_mem_write8
        ret     c
        ld      hl, (bios_clear_address)
        inc     hl
        ld      (bios_clear_address), hl
        ld      hl, (bios_clear_remaining)
        dec     hl
        ld      (bios_clear_remaining), hl
        ld      a, h
        or      l
        jr      nz, .bios_clear_loop
        or      a
        ret

; Verify all 1024 bytes of the IVT are zero.
bios_verify_ivt:
        ld      hl, BIOS_IVT_BASE
        ld      (bios_verify_address), hl
        ld      hl, BIOS_IVT_SIZE
        ld      (bios_verify_remaining), hl
.bios_verify_loop:
        ld      hl, (bios_verify_address)
        ld      (x86_mem_addr), hl
        xor     a
        ld      (x86_mem_addr+2), a
        call    x86_mem_read8
        ret     c
        or      a
        jr      nz, .bios_verify_failed
        ld      hl, (bios_verify_address)
        inc     hl
        ld      (bios_verify_address), hl
        ld      hl, (bios_verify_remaining)
        dec     hl
        ld      (bios_verify_remaining), hl
        ld      a, h
        or      l
        jr      nz, .bios_verify_loop
        or      a
        ret
.bios_verify_failed:
        scf
        ret

; IN: HL=16-bit physical address, A=value. Carry on memory error.
bios_write_byte:
        ld      (bios_work_value), a
        ld      (x86_mem_addr), hl
        xor     a
        ld      (x86_mem_addr+2), a
        ld      a, (bios_work_value)
        ld      (x86_mem_value), a
        jp      x86_mem_write8

; IN: HL=16-bit physical address. OUT: A=value, carry on memory error.
bios_read_byte:
        ld      (x86_mem_addr), hl
        xor     a
        ld      (x86_mem_addr+2), a
        jp      x86_mem_read8

; IN: HL=address, A=expected byte. Carry if read fails or value differs.
bios_expect_byte:
        ld      (bios_work_value), a
        call    bios_read_byte
        ret     c
        ld      b, a
        ld      a, (bios_work_value)
        cp      b
        ret     z
        scf
        ret

bios_post_fail_1:
        ld      a, 1
        jr      bios_post_fail
bios_post_fail_2:
        ld      a, 2
        jr      bios_post_fail
bios_post_fail_3:
        ld      a, 3
        jr      bios_post_fail
bios_post_fail_4:
        ld      a, 4
        jr      bios_post_fail
bios_post_fail_5:
        ld      a, 5
        jr      bios_post_fail
bios_post_fail_6:
        ld      a, 6
        jr      bios_post_fail
bios_post_fail_7:
        ld      a, 7
        jr      bios_post_fail
bios_post_fail_8:
        ld      a, 8

bios_post_fail:
        ld      l, a
        ld      h, #B1
        ld      (saved_x86_ax), hl
        ld      a, 7
        ld      (x86_exit_status), a
        scf
        ret

bios_clear_address:   dw 0
bios_clear_remaining: dw 0
bios_verify_address:  dw 0
bios_verify_remaining:dw 0
bios_work_value:      db 0

        ASSERT BIOS_IVT_BASE + BIOS_IVT_SIZE = BIOS_BDA_BASE
        ASSERT BIOS_BDA_BASE + BIOS_BDA_SIZE = BIOS_LOW_MEMORY_END


; =========================================================================
; FILE: neogs_sub/bios/bios_video.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — BIOS INT 10h text services, Stage 3A
;
; Project text-console profile: 42 columns x 12 rows, 6x8 glyphs.
; Implemented functions:
;   AH=00h set mode 03h (project 42x12 text profile)
;   AH=0Fh get mode (AL=03h, AH=42, BH=0)
;   AH=02h set cursor (BH=0, DH=row, DL=column)
;   AH=03h get cursor (DH=row, DL=column)
;   AH=0Eh teletype output (AL=character, BH=0)
;
; INT 10h enters through the normal IVT/stack frame. The BIOS service itself
; is a firmware trap, then returns through the normal IRET primitive.
; Host stream protocol over DBIT/GSDAT:
;   nonzero byte = teletype character
;   00 00        = clear the lower debug console
;   00 01 row col= set lower debug console cursor
; =========================================================================

BIOS_VIDEO_COLUMNS EQU 42
BIOS_VIDEO_ROWS    EQU 12
BIOS_VIDEO_MODE    EQU #03

bios_int10_dispatch:
        ld      hl, (x86_reg_ax)
        ld      a, h
        cp      #00
        jp      z, bios_int10_set_mode
        cp      #0F
        jp      z, bios_int10_get_mode
        cp      #02
        jp      z, bios_int10_set_cursor
        cp      #03
        jp      z, bios_int10_get_cursor
        cp      #0E
        jp      z, bios_int10_teletype
        or      a
        ret

; Install IVT[10h] -> 0000:5200 (firmware service trap).
bios_int10_install_vector:
        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0040
        ld      (x86_mem_offset), hl
        ld      hl, #5200
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        ret     c
        ld      hl, 0

; =========================================================================
; FILE: neogs_sub/x86_interrupt.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — 8086 real-mode interrupt frame engine (Stage 2A)
;
; Implements the architectural memory/stack part of software INT and IRET.
; This module is deliberately separate from the legacy four-opcode fetch
; loop: opcode dispatch and execution of guest handler code are Stage 2B.
;
; x86_int_enter:
;   IN: A = interrupt vector number
;   Uses x86_reg_{ss,sp,cs,ip,flags}.
;   Push order: FLAGS, CS, IP; final SP points to saved IP.
;   Loads new IP:CS from IVT[vector], clears IF and TF.
;   OUT: carry clear on success, carry set on memory error.
;
; x86_iret_return:
;   Pops IP, CS, FLAGS; restores all three and advances SP by six.
;
; x86_int_self_test:
;   Installs a temporary vector 20h, validates the exact stack frame,
;   validates IF/TF clearing, executes IRET, validates restored state,
;   and removes the temporary vector.
;   Success: status=0, AX=1A01h. Failure: status=8, AX=1Axxh.
; =========================================================================

X86_INT_IF_TF_CLEAR_MASK EQU #FC

x86_int_enter:
        ld      (x86_int_vector), a

        ; IVT entry address = vector * 4, physical segment zero.
        ld      l, a
        ld      h, 0
        add     hl, hl
        add     hl, hl
        ld      (x86_mem_offset), hl
        ld      hl, 0
        ld      (x86_mem_segment), hl
        call    x86_mem_read16
        ret     c
        ld      (x86_int_target_ip), hl

        ld      hl, (x86_mem_offset)
        inc     hl
        inc     hl
        ld      (x86_mem_offset), hl
        ld      hl, 0
        ld      (x86_mem_segment), hl
        call    x86_mem_read16
        ret     c
        ld      (x86_int_target_cs), hl

        ; 8086 interrupt entry pushes FLAGS, then CS, then return IP.
        ld      de, (x86_reg_flags)
        call    x86_int_push_word
        ret     c
        ld      de, (x86_reg_cs)
        call    x86_int_push_word
        ret     c
        ld      de, (x86_reg_ip)
        call    x86_int_push_word
        ret     c

        ; IF (bit 9) and TF (bit 8) are cleared in the live FLAGS register.
        ld      a, (x86_reg_flags+1)
        and     X86_INT_IF_TF_CLEAR_MASK
        ld      (x86_reg_flags+1), a
        ld      hl, (x86_int_target_ip)
        ld      (x86_reg_ip), hl
        ld      hl, (x86_int_target_cs)
        ld      (x86_reg_cs), hl
        or      a
        ret

x86_iret_return:
        call    x86_int_pop_word
        ret     c
        ld      (x86_reg_ip), hl

        call    x86_int_pop_word
        ret     c
        ld      (x86_reg_cs), hl

        call    x86_int_pop_word
        ret     c
        ld      (x86_reg_flags), hl
        or      a
        ret

; IN: DE=word. Decrement guest SP by two, then write SS:SP.
x86_int_push_word:
        ld      (x86_int_word), de
        ld      hl, (x86_reg_sp)
        dec     hl
        dec     hl
        ld      (x86_reg_sp), hl
        ld      (x86_mem_offset), hl
        ld      de, (x86_reg_ss)
        ld      (x86_mem_segment), de
        ld      hl, (x86_int_word)
        ld      (x86_mem_word_value), hl
        jp      x86_mem_write16

; OUT: HL=word at SS:SP. SP advances only after a successful read.
x86_int_pop_word:
        ld      hl, (x86_reg_sp)
        ld      (x86_mem_offset), hl
        ld      de, (x86_reg_ss)
        ld      (x86_mem_segment), de
        call    x86_mem_read16
        ret     c
        push    hl
        ld      hl, (x86_reg_sp)
        inc     hl
        inc     hl
        ld      (x86_reg_sp), hl
        pop     hl
        or      a
        ret

; -------------------------------------------------------------------------
; Deterministic architectural test. The test does not execute guest handler
; instructions; it validates the real-mode entry/return frame primitives.
; -------------------------------------------------------------------------
x86_int_self_test:
        ; Temporary IVT[20h] = 5678:1234.
        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0080
        ld      (x86_mem_offset), hl
        ld      hl, #1234
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        jp      c, x86_int_test_fail_1

        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0082
        ld      (x86_mem_offset), hl
        ld      hl, #5678
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        jp      c, x86_int_test_fail_1

        ; Known caller state. IF and TF are both set in FLAGS=0302h.
        ld      hl, #7000
        ld      (x86_reg_ss), hl
        ld      hl, #0100
        ld      (x86_reg_sp), hl
        ld      hl, #1111
        ld      (x86_reg_cs), hl
        ld      hl, #2222
        ld      (x86_reg_ip), hl
        ld      hl, #0302
        ld      (x86_reg_flags), hl

        ld      a, #20
        call    x86_int_enter
        jp      c, x86_int_test_fail_2

        ; Verify target and the post-entry architectural state.
        ld      hl, (x86_reg_ip)
        ld      de, #1234
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_3
        ld      hl, (x86_reg_cs)
        ld      de, #5678
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_3
        ld      hl, (x86_reg_sp)
        ld      de, #00FA
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_3
        ld      hl, (x86_reg_flags)
        ld      de, #0002
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_3

        ; Stack layout at SS:SP must be IP, CS, FLAGS (top to bottom).
        ld      hl, #7000
        ld      (x86_mem_segment), hl
        ld      hl, #00FA
        ld      (x86_mem_offset), hl
        call    x86_mem_read16
        jp      c, x86_int_test_fail_4
        ld      de, #2222
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_4

        ld      hl, #7000
        ld      (x86_mem_segment), hl
        ld      hl, #00FC
        ld      (x86_mem_offset), hl
        call    x86_mem_read16
        jp      c, x86_int_test_fail_4
        ld      de, #1111
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_4

        ld      hl, #7000
        ld      (x86_mem_segment), hl
        ld      hl, #00FE
        ld      (x86_mem_offset), hl
        call    x86_mem_read16
        jp      c, x86_int_test_fail_4
        ld      de, #0302
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_4

        ; IRET must restore IP, CS, FLAGS and the original SP.
        call    x86_iret_return
        jp      c, x86_int_test_fail_5
        ld      hl, (x86_reg_ip)
        ld      de, #2222
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_5
        ld      hl, (x86_reg_cs)
        ld      de, #1111
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_5
        ld      hl, (x86_reg_flags)
        ld      de, #0302
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_5
        ld      hl, (x86_reg_sp)
        ld      de, #0100
        or      a
        sbc     hl, de
        jp      nz, x86_int_test_fail_5

        ; Remove temporary vector so BIOS IVT remains uninstalled.
        call    x86_int_clear_test_vector
        jp      c, x86_int_test_fail_6

        ld      hl, #1A01
        ld      (saved_x86_ax), hl
        xor     a
        ld      (x86_exit_status), a
        or      a
        ret

x86_int_clear_test_vector:
        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0080
        ld      (x86_mem_offset), hl
        ld      hl, 0
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        ret     c
        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0082
        ld      (x86_mem_offset), hl
        ld      hl, 0
        ld      (x86_mem_word_value), hl
        jp      x86_mem_write16

x86_int_test_fail_1:
        ld      a, 1
        jr      x86_int_test_fail
x86_int_test_fail_2:
        ld      a, 2
        jr      x86_int_test_fail
x86_int_test_fail_3:
        ld      a, 3
        jr      x86_int_test_fail
x86_int_test_fail_4:
        ld      a, 4
        jr      x86_int_test_fail
x86_int_test_fail_5:
        ld      a, 5
        jr      x86_int_test_fail
x86_int_test_fail_6:

x86_int_test_fail:
        ld      l, a
        ld      h, #1A
        ld      (saved_x86_ax), hl
        ld      a, 8
        ld      (x86_exit_status), a
        scf
        ret

x86_int_vector:    db 0
x86_int_word:      dw 0
x86_int_target_ip: dw 0
x86_int_target_cs: dw 0


; =========================================================================
; FILE: neogs_sub/x86_real_mode.asm
; =========================================================================
; =========================================================================
; PROJECT 124 — 8086 real-mode fetch/execute core, Stage 2B
;
; This first executable slice runs guest instructions through CS:IP and the
; physical-memory backend. Supported: NOP, INC AX, MOV r8/r16,imm, HLT,
; INT imm8, INT3, IRET. It is intentionally not a general 8086 emulator.
;
; x86_real_mode_test installs a guest program at 0000:5002 and an interrupt
; handler at 0000:5100, then executes:
;   MOV AX,0000 / INT 20h / HLT
; handler:
;   MOV AX,1234 / IRET
; The test checks AX, CS:IP return path, FLAGS and SP balance.
; Success: status=0, AX=1A02h. Failure: status=9, AX=1Bxxh.
; =========================================================================

X86_REAL_STEP_LIMIT EQU #FFFF

x86_real_mode_test:
        xor     a
        ld      (x86_real_bounds_enable), a
        call    x86_mem_init

        ; Copy main guest program to physical 0000:5002.
        ld      hl, x86_guest_main_image
        ld      de, #5002
        ld      bc, x86_guest_main_image_end - x86_guest_main_image
        call    x86_real_copy_image
        jp      c, x86_real_test_fail_1

        ; Copy interrupt handler to physical 0000:5100.
        ld      hl, x86_guest_handler_image
        ld      de, #5100
        ld      bc, x86_guest_handler_image_end - x86_guest_handler_image
        call    x86_real_copy_image
        jp      c, x86_real_test_fail_1

        ; Install IVT[20h] = 0000:5100.
        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0080
        ld      (x86_mem_offset), hl
        ld      hl, #5100
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        jp      c, x86_real_test_fail_2
        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0082
        ld      (x86_mem_offset), hl
        ld      hl, 0
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        jp      c, x86_real_test_fail_2

        ; Initialize guest architectural state.
        ld      hl, 0
        ld      (x86_reg_ax), hl
        ld      (x86_reg_cx), hl
        ld      (x86_reg_dx), hl
        ld      (x86_reg_bx), hl
        ld      (x86_reg_bp), hl
        ld      (x86_reg_si), hl
        ld      (x86_reg_di), hl
        ld      (x86_reg_es), hl
        ld      (x86_reg_ds), hl
        ld      (x86_reg_cs), hl
        ld      hl, #5002
        ld      (x86_reg_ip), hl
        ld      hl, #7000
        ld      (x86_reg_ss), hl
        ld      hl, #0100
        ld      (x86_reg_sp), hl
        ld      hl, #0302
        ld      (x86_reg_flags), hl

        call    x86_real_execute
        jp      c, x86_real_test_fail_3

        ; Handler's MOV AX,1234 must execute before guest IRET returns to HLT.
        ld      hl, (x86_reg_ax)
        ld      de, #1234
        or      a
        sbc     hl, de
        jp      nz, x86_real_test_fail_4

        ; Verify original control-flow context and stack were restored.
        ld      hl, (x86_reg_cs)
        ld      de, 0
        or      a
        sbc     hl, de
        jp      nz, x86_real_test_fail_5
        ld      hl, (x86_reg_ip)
        ld      de, #5008
        or      a
        sbc     hl, de
        jp      nz, x86_real_test_fail_5
        ld      hl, (x86_reg_sp)
        ld      de, #0100
        or      a
        sbc     hl, de
        jp      nz, x86_real_test_fail_5
        ld      hl, (x86_reg_flags)
        ld      de, #0302
        or      a
        sbc     hl, de
        jp      nz, x86_real_test_fail_5

        call    x86_real_clear_vector
        jp      c, x86_real_test_fail_6

        ld      hl, #1A02
        ld      (saved_x86_ax), hl
        xor     a
        ld      (x86_exit_status), a
        or      a
        ret

; IN: HL=source in firmware, DE=physical x86 destination, BC=length.
x86_real_copy_image:
        ld      (x86_real_copy_source), hl
        ld      (x86_real_copy_dest), de
        ld      (x86_real_copy_count), bc
x86_real_copy_loop:
        ld      bc, (x86_real_copy_count)
        ld      a, b
        or      c
        ret     z
        ld      hl, (x86_real_copy_source)
        ld      a, (hl)
        ld      (x86_mem_value), a
        ld      hl, (x86_real_copy_dest)
        ld      (x86_mem_addr), hl
        xor     a
        ld      (x86_mem_addr+2), a
        call    x86_mem_write8
        ret     c
        ld      hl, (x86_real_copy_source)
        inc     hl
        ld      (x86_real_copy_source), hl
        ld      hl, (x86_real_copy_dest)
        inc     hl
        ld      (x86_real_copy_dest), hl
        ld      hl, (x86_real_copy_count)
        dec     hl
        ld      (x86_real_copy_count), hl
        jr      x86_real_copy_loop

; Fetch one guest byte from physical ((CS << 4) + IP) & FFFFFh.
; OUT A=byte; guest IP advances only after a successful memory read.
x86_real_fetch8:
        ld      a, (x86_real_bounds_enable)
        or      a
        jr      z, .fetch_unbounded
        ld      hl, (x86_reg_cs)
        ld      a, h
        or      l
        jp      nz, x86_real_fetch_out_of_range
        ld      hl, (x86_reg_ip)
        ld      de, #5002
        or      a
        sbc     hl, de
        jp      c, x86_real_fetch_out_of_range
        ld      hl, (x86_real_task_end)
        ld      de, (x86_reg_ip)
        or      a
        sbc     hl, de
        jp      c, x86_real_fetch_out_of_range
        ld      a, h
        or      l
        jp      z, x86_real_fetch_out_of_range
.fetch_unbounded:
        ld      hl, (x86_reg_ip)
        ld      (x86_mem_offset_work), hl
        ld      hl, (x86_reg_cs)
        ld      (x86_mem_segment), hl
        call    x86_mem_calc_phys
        call    x86_mem_read8
        ret     c
        push    af
        ld      hl, (x86_reg_ip)
        inc     hl
        ld      (x86_reg_ip), hl
        pop     af
        or      a
        ret

x86_real_fetch_out_of_range:
        ld      a, 4
        ld      (x86_exit_status), a
        scf
        ret

x86_real_execute:
        ld      hl, X86_REAL_STEP_LIMIT
        ld      (x86_real_steps), hl
x86_real_next:
        ld      hl, (x86_real_steps)
        ld      a, h
        or      l
        jp      z, x86_real_budget_fail
        dec     hl
        ld      (x86_real_steps), hl

        call    x86_real_fetch8
        ret     c

        cp      #90
        jp      z, x86_real_nop
        cp      #40
        jp      z, x86_real_inc_ax
        cp      #B0
        jp      c, .not_mov_r8
        cp      #B8
        jp      c, x86_real_mov_r8_imm8
.not_mov_r8:
        cp      #B8
        jp      z, x86_real_mov_ax_imm16
        cp      #BA
        jp      z, x86_real_mov_dx_imm16
        cp      #BB
        jp      z, x86_real_mov_bx_imm16
        cp      #F4
        jp      z, x86_real_hlt
        cp      #CD
        jp      z, x86_real_int_imm8
        cp      #CC
        jp      z, x86_real_int3
        cp      #CF
        jp      z, x86_real_iret

        ld      a, 2
        ld      (x86_exit_status), a
        scf
        ret

x86_real_nop:
        jp      x86_real_next

x86_real_inc_ax:
        ld      hl, (x86_reg_ax)
        inc     hl
        ld      (x86_reg_ax), hl
        jp      x86_real_next

x86_real_mov_ax_imm16:
        call    x86_real_fetch8
        ret     c
        ld      (x86_real_imm_low), a
        call    x86_real_fetch8
        ret     c
        ld      h, a
        ld      a, (x86_real_imm_low)
        ld      l, a
        ld      (x86_reg_ax), hl
        jp      x86_real_next

; B0-B7 — MOV r8,imm8 (AL,CL,DL,BL,AH,CH,DH,BH).
x86_real_mov_r8_imm8:
        ld      (x86_real_opcode), a
        call    x86_real_fetch8
        ret     c
        ld      (x86_real_imm_byte), a
        ld      a, (x86_real_opcode)
        sub     #B0
        cp      4
        jr      c, .low_byte_register
        sub     4
        add     a, a
        inc     a
        jr      .register_offset_ready
.low_byte_register:
        add     a, a
.register_offset_ready:
        ld      e, a
        ld      d, 0
        ld      hl, x86_reg_ax
        add     hl, de
        ld      a, (x86_real_imm_byte)
        ld      (hl), a
        jp      x86_real_next

x86_real_int_imm8:
        call    x86_real_fetch8
        ret     c
        ld      (x86_real_int_vector), a
        call    x86_int_enter
        ret     c
        ld      a, (x86_real_int_vector)
        cp      #10
        jp      nz, x86_real_next
        call    bios_int10_dispatch
        ret     c
        call    x86_iret_return
        ret     c
        jp      x86_real_next

x86_real_mov_bx_imm16:
        call    x86_real_fetch8
        ret     c
        ld      (x86_real_imm_low), a
        call    x86_real_fetch8
        ret     c
        ld      h, a
        ld      a, (x86_real_imm_low)
        ld      l, a
        ld      (x86_reg_bx), hl
        jp      x86_real_next

x86_real_mov_dx_imm16:
        call    x86_real_fetch8
        ret     c
        ld      (x86_real_imm_low), a
        call    x86_real_fetch8
        ret     c
        ld      h, a
        ld      a, (x86_real_imm_low)
        ld      l, a
        ld      (x86_reg_dx), hl
        jp      x86_real_next

x86_real_int3:
        ld      a, 3
        call    x86_int_enter
        ret     c
        jp      x86_real_next

x86_real_iret:
        call    x86_iret_return
        ret     c
        jp      x86_real_next

x86_real_hlt:
        xor     a
        ld      (x86_exit_status), a
        or      a
        ret

x86_real_budget_fail:
        ld      a, 1
        ld      (x86_exit_status), a
        scf
        ret

x86_real_clear_vector:
        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0080
        ld      (x86_mem_offset), hl
        ld      hl, 0
        ld      (x86_mem_word_value), hl
        call    x86_mem_write16
        ret     c
        ld      hl, 0
        ld      (x86_mem_segment), hl
        ld      hl, #0082
        ld      (x86_mem_offset), hl
        ld      hl, 0
        ld      (x86_mem_word_value), hl
        jp      x86_mem_write16

x86_real_test_fail_1:
        ld      a, 1
        jr      x86_real_test_fail
x86_real_test_fail_2:
        ld      a, 2
        jr      x86_real_test_fail
x86_real_test_fail_3:
        ld      a, 3
        jr      x86_real_test_fail
x86_real_test_fail_4:
        ld      a, 4
        jr      x86_real_test_fail
x86_real_test_fail_5:
        ld      a, 5
        jr      x86_real_test_fail
x86_real_test_fail_6:
        ld      a, 6
x86_real_test_fail:
        ld      l, a
        ld      h, #1B
        ld      (saved_x86_ax), hl
        ld      a, 9
        ld      (x86_exit_status), a
        scf
        ret

x86_real_steps:       dw 0
x86_real_copy_source: dw 0
x86_real_copy_dest:   dw 0
x86_real_copy_count:  dw 0
x86_real_imm_low:    db 0
x86_real_int_vector: db 0
x86_real_opcode: db 0
x86_real_imm_byte: db 0
x86_real_bounds_enable: db 0
x86_real_task_end: dw 0

x86_guest_main_image:
        db #B8,#00,#00,#CD,#20,#F4
x86_guest_main_image_end:
x86_guest_handler_image:
        db #B8,#34,#12,#CF
x86_guest_handler_image_end:

; FILE: neogs_sub/build_firmware.asm
; =========================================================================
; =========================================================================
; Build both NeoGS binaries from the Project 124 root:
;   sjasmplus neogs_sub/build_firmware.asm
;
; neogs_code.bin       custom firmware, loaded at logical RAM address $0000
; neogs_bootstrap.bin  executable Z80 bootstrap, loaded by #14 at $4000
; =========================================================================
        DEVICE ZXSPECTRUM128
        INCLUDE "main.asm"

; Low RAM is used by the firmware and its stack. Keep the image below $3000;
; the Z80 stack is placed at $3FFF and the x86 DMA payload starts at $5000.
        ASSERT neogs_firmware_end <= $3000
        SAVEBIN "neogs_code.bin", 0, neogs_firmware_end
        SAVEBIN "zx_main/neogs_code.bin", 0, neogs_firmware_end

; GS105a #14 writes the executable bootstrap into fixed RAM $4000-$7FFF.
        ORG     $4000
        INCLUDE "bootstrap.asm"
        ASSERT neogs_bootstrap_start = $4000
        ASSERT neogs_bootstrap_end <= $7FFF
        SAVEBIN "neogs_bootstrap.bin", neogs_bootstrap_start, neogs_bootstrap_end - neogs_bootstrap_start
        SAVEBIN "zx_main/neogs_bootstrap.bin", neogs_bootstrap_start, neogs_bootstrap_end - neogs_bootstrap_start
